// decls prints one line per top-level declaration of the given Go files:
// kind, name and a SHA-256 of its exact source text (doc comment included).
// Import declarations are skipped. Comments that belong to no declaration
// are printed as "comment". Sorting the output of two file sets and
// comparing them shows whether a split was a pure move.
package main

import (
	"crypto/sha256"
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"sort"
	"strings"
)

func main() {
	var lines []string
	for _, path := range os.Args[1:] {
		src, err := os.ReadFile(path)
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(2)
		}
		fset := token.NewFileSet()
		file, err := parser.ParseFile(fset, path, src, parser.ParseComments)
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(2)
		}
		type span struct{ start, end int }
		var spans []span
		text := func(start, end token.Pos) string {
			s, e := fset.Position(start).Offset, fset.Position(end).Offset
			spans = append(spans, span{s, e})
			return string(src[s:e])
		}
		emit := func(kind, name, body string) {
			sum := sha256.Sum256([]byte(body))
			lines = append(lines, fmt.Sprintf("%s %s %x %d", kind, name, sum[:8], strings.Count(body, "\n")+1))
		}
		for _, decl := range file.Decls {
			switch d := decl.(type) {
			case *ast.FuncDecl:
				start := d.Pos()
				if d.Doc != nil {
					start = d.Doc.Pos()
				}
				name := d.Name.Name
				if d.Recv != nil && len(d.Recv.List) > 0 {
					name = recvName(d.Recv.List[0].Type) + "." + name
				}
				emit("func", name, text(start, d.End()))
			case *ast.GenDecl:
				if d.Tok == token.IMPORT {
					spans = append(spans, span{fset.Position(d.Pos()).Offset, fset.Position(d.End()).Offset})
					continue
				}
				start := d.Pos()
				if d.Doc != nil {
					start = d.Doc.Pos()
				}
				var names []string
				for _, spec := range d.Specs {
					switch s := spec.(type) {
					case *ast.TypeSpec:
						names = append(names, s.Name.Name)
					case *ast.ValueSpec:
						for _, n := range s.Names {
							names = append(names, n.Name)
						}
					}
				}
				emit(d.Tok.String(), strings.Join(names, ","), text(start, d.End()))
			}
		}
		for _, group := range file.Comments {
			s, e := fset.Position(group.Pos()).Offset, fset.Position(group.End()).Offset
			inside := false
			for _, sp := range spans {
				if s >= sp.start && e <= sp.end {
					inside = true
					break
				}
			}
			if inside || group.Pos() < file.Package {
				continue
			}
			emit("comment", "-", string(src[s:e]))
		}
	}
	sort.Strings(lines)
	for _, line := range lines {
		fmt.Println(line)
	}
}

func recvName(expr ast.Expr) string {
	switch t := expr.(type) {
	case *ast.StarExpr:
		return recvName(t.X)
	case *ast.Ident:
		return t.Name
	case *ast.IndexExpr:
		return recvName(t.X)
	}
	return "?"
}
