// movedecls moves top-level declarations out of one Go file into other files
// of the same package, byte for byte.
//
// Usage: go run main.go SOURCE.go MAPPING
//
// MAPPING is a text file. Each non-empty line that does not start with '#'
// reads "target.go: name name ...". A function is named by its name, a
// method by "Recv.Method", a const/var/type declaration by the first name it
// declares. Every declaration not named stays in SOURCE.go.
//
// A declaration moves with its doc comment and with a trailing comment on
// its last line. Declarations keep their source order inside each target. A
// new target gets the package clause and a copy of the source's whole import
// block (run goimports afterwards to drop the unused ones); an existing
// target is appended to. A comment group that belongs to no declaration
// stays in SOURCE.go and is reported on stderr.
package main

import (
	"bytes"
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"path/filepath"
	"strings"
)

type decl struct {
	name       string
	start, end int // byte offsets; end is exclusive and includes the newline
}

func main() {
	if len(os.Args) != 3 {
		fail("usage: movedecls SOURCE.go MAPPING")
	}
	srcPath := os.Args[1]
	src, err := os.ReadFile(srcPath)
	check(err)
	targetOf, targets := readMapping(os.Args[2])

	fset := token.NewFileSet()
	file, err := parser.ParseFile(fset, srcPath, src, parser.ParseComments)
	check(err)
	off := func(p token.Pos) int { return fset.Position(p).Offset }

	var importBlock []byte
	var decls []decl
	for _, d := range file.Decls {
		start, end := d.Pos(), d.End()
		var name string
		switch t := d.(type) {
		case *ast.FuncDecl:
			if t.Doc != nil {
				start = t.Doc.Pos()
			}
			name = t.Name.Name
			if t.Recv != nil && len(t.Recv.List) > 0 {
				name = recvName(t.Recv.List[0].Type) + "." + name
			}
		case *ast.GenDecl:
			if t.Tok == token.IMPORT {
				importBlock = append(importBlock, src[off(start):off(end)]...)
				importBlock = append(importBlock, '\n')
				continue
			}
			if t.Doc != nil {
				start = t.Doc.Pos()
			}
			name = firstName(t)
		}
		s, e := off(start), off(end)
		// Take the rest of the last line (a trailing comment) and its newline.
		if nl := bytes.IndexByte(src[e:], '\n'); nl >= 0 {
			e += nl + 1
		} else {
			e = len(src)
		}
		// Start at the beginning of the line.
		for s > 0 && src[s-1] != '\n' {
			s--
		}
		decls = append(decls, decl{name: name, start: s, end: e})
	}

	seen := map[string]int{}
	for _, d := range decls {
		seen[d.name]++
	}
	for name := range targetOf {
		switch seen[name] {
		case 0:
			fail("mapping names %q, which %s does not declare", name, srcPath)
		case 1:
		default:
			fail("%q is declared %d times in %s; cannot tell which to move", name, seen[name], srcPath)
		}
	}

	for _, group := range file.Comments {
		s, e := off(group.Pos()), off(group.End())
		if group.Pos() < file.Package {
			continue
		}
		inside := false
		for _, d := range decls {
			if s >= d.start && e <= d.end {
				inside = true
				break
			}
		}
		if !inside && !bytes.Contains(importBlock, src[s:e]) {
			fmt.Fprintf(os.Stderr, "note: free comment stays in the source at line %d\n", fset.Position(group.Pos()).Line)
		}
	}

	moved := map[string][][]byte{}
	var kept bytes.Buffer
	pos := 0
	for _, d := range decls {
		target, ok := targetOf[d.name]
		if !ok {
			continue
		}
		kept.Write(src[pos:d.start])
		moved[target] = append(moved[target], src[d.start:d.end])
		pos = d.end
		// Drop one blank line after a moved declaration.
		if pos < len(src) && src[pos] == '\n' {
			pos++
		}
	}
	kept.Write(src[pos:])

	dir := filepath.Dir(srcPath)
	for _, target := range targets {
		path := filepath.Join(dir, target)
		var out bytes.Buffer
		existing, err := os.ReadFile(path)
		switch {
		case err == nil:
			out.Write(bytes.TrimRight(existing, "\n"))
			out.WriteString("\n")
		case os.IsNotExist(err):
			fmt.Fprintf(&out, "package %s\n\n", file.Name.Name)
			out.Write(importBlock)
		default:
			check(err)
		}
		for _, body := range moved[target] {
			out.WriteString("\n")
			out.Write(body)
		}
		check(os.WriteFile(path, out.Bytes(), 0o644))
		fmt.Printf("%s: %d declarations\n", target, len(moved[target]))
	}
	check(os.WriteFile(srcPath, bytes.TrimRight(kept.Bytes(), "\n"), 0o644))
	f, err := os.OpenFile(srcPath, os.O_APPEND|os.O_WRONLY, 0o644)
	check(err)
	_, err = f.WriteString("\n")
	check(err)
	check(f.Close())
	fmt.Printf("%s: %d declarations kept\n", filepath.Base(srcPath), len(decls)-len(targetOf))
}

func readMapping(path string) (map[string]string, []string) {
	data, err := os.ReadFile(path)
	check(err)
	targetOf := map[string]string{}
	var targets []string
	for _, line := range strings.Split(string(data), "\n") {
		line = strings.TrimSpace(line)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		target, names, ok := strings.Cut(line, ":")
		if !ok {
			fail("mapping line without a colon: %q", line)
		}
		target = strings.TrimSpace(target)
		targets = append(targets, target)
		for _, name := range strings.Fields(names) {
			if prev, dup := targetOf[name]; dup {
				fail("%q is mapped to both %s and %s", name, prev, target)
			}
			targetOf[name] = target
		}
	}
	return targetOf, targets
}

func firstName(d *ast.GenDecl) string {
	for _, spec := range d.Specs {
		switch s := spec.(type) {
		case *ast.TypeSpec:
			return s.Name.Name
		case *ast.ValueSpec:
			if len(s.Names) > 0 {
				return s.Names[0].Name
			}
		}
	}
	return "?"
}

func recvName(expr ast.Expr) string {
	switch t := expr.(type) {
	case *ast.StarExpr:
		return recvName(t.X)
	case *ast.Ident:
		return t.Name
	case *ast.IndexExpr:
		return recvName(t.X)
	}
	return "?"
}

func check(err error) {
	if err != nil {
		fail("%v", err)
	}
}

func fail(format string, args ...any) {
	fmt.Fprintf(os.Stderr, format+"\n", args...)
	os.Exit(2)
}
