// unexport gives back their unexported names to the identifiers of the sync
// tree that were exported only so other packages-to-be could use them and that,
// after the split, no other package uses.
//
//	go -C <repo> run /path/unexport.go -repo <repo> -base <commit before the exports> [-dry]
//
// A candidate is an exported declaration, method or field of a package under
// internal/sync whose name was unexported at -base (same owner, first letter
// lower case) and is exported now. It is un-exported only when
//
//   - no other package names it: every package of the module that imports the
//     sync tree is type-checked with its tests, for the Linux and the macOS
//     file sets, and with the e2e tags;
//   - it is not a type reachable from the signature or fields of an object
//     another package names;
//   - it is not a method of a name any interface of a loaded package declares,
//     and not an interface method (interface satisfaction is left alone);
//   - no string literal of the module names it (source-reading tests), and its
//     lower-case name resolves to nothing at any of its use sites, nor is a
//     field or method of its type already.
//
// Every definition and use is renamed, with the first word of the doc comment
// that names the identifier, and backticked mentions in spec/design and
// spec/reference when no identifier that stays exported has the same name.
// Every written Go file is gofmt-formatted. Standard library only.
package main

var (
	repoFlag = flag.String("repo", ".", "repository root")
	baseFlag = flag.String("base", "", "commit at which the candidates were still unexported")
	dryFlag  = flag.Bool("dry", false, "print the candidates and the reasons kept, write nothing")
)

const (
	syncRel = "internal/sync"
	tags    = "e2e e2e_full integration"
)

func fail(format string, a ...any) {
	fmt.Fprintf(os.Stderr, "unexport: "+format+"\n", a...)
	os.Exit(1)
}

func lowerFirst(s string) string {
	r, n := utf8.DecodeRuneInString(s)
	return string(unicode.ToLower(r)) + s[n:]
}

type goPkg struct {
	Dir, ImportPath, Name              string
	GoFiles, TestGoFiles, XTestGoFiles []string
	Imports, TestImports               []string
}

func gitCmd(args ...string) []byte { return run("git", nil, args...) }

func goCmd(env []string, args ...string) []byte { return run("go", env, args...) }

func run(tool string, env []string, args ...string) []byte {
	cmd := exec.Command(tool, args...)
	cmd.Dir = *repoFlag
	cmd.Env = append(os.Environ(), env...)
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	out, err := cmd.Output()
	if err != nil {
		fail("%s %s: %v\n%s", tool, strings.Join(args, " "), err, stderr.String())
	}
	return out
}

// ---------- the names at -base ----------

// baseNames are (lower-cased owner, name) pairs declared under internal/sync
// at -base: owner "" for a package-level name, else the receiver or struct or
// interface type.
func baseNames() map[[2]string]bool {
	out := gitCmd("ls-tree", "-r", "--name-only", *baseFlag, syncRel+"/")
	names := map[[2]string]bool{}
	fset := token.NewFileSet()
	for _, path := range strings.Fields(string(out)) {
		if !strings.HasSuffix(path, ".go") {
			continue
		}
		src := gitCmd("show", *baseFlag+":"+path)
		f, err := parser.ParseFile(fset, path, src, parser.SkipObjectResolution)
		if err != nil {
			fail("parse %s at base: %v", path, err)
		}
		for _, d := range f.Decls {
			switch x := d.(type) {
			case *ast.FuncDecl:
				owner := ""
				if x.Recv != nil && len(x.Recv.List) > 0 {
					owner = strings.ToLower(recvBase(x.Recv.List[0].Type))
				}
				names[[2]string{owner, x.Name.Name}] = true
			case *ast.GenDecl:
				for _, sp := range x.Specs {
					switch s := sp.(type) {
					case *ast.TypeSpec:
						names[[2]string{"", s.Name.Name}] = true
						owner := strings.ToLower(s.Name.Name)
						switch t := s.Type.(type) {
						case *ast.StructType:
							for _, fl := range t.Fields.List {
								for _, n := range fl.Names {
									names[[2]string{owner, n.Name}] = true
								}
							}
						case *ast.InterfaceType:
							for _, m := range t.Methods.List {
								for _, n := range m.Names {
									names[[2]string{owner, n.Name}] = true
								}
							}
						}
					case *ast.ValueSpec:
						for _, n := range s.Names {
							names[[2]string{"", n.Name}] = true
						}
					}
				}
			}
		}
	}
	return names
}

func recvBase(e ast.Expr) string {
	for {
		switch x := e.(type) {
		case *ast.StarExpr:
			e = x.X
		case *ast.ParenExpr:
			e = x.X
		case *ast.IndexExpr:
			e = x.X
		case *ast.IndexListExpr:
			e = x.X
		case *ast.Ident:
			return x.Name
		default:
			return ""
		}
	}
}

func recvNamed(t types.Type) *types.TypeName {
	for {
		switch u := t.(type) {
		case *types.Pointer:
			t = u.Elem()
			continue
		case *types.Alias:
			t = types.Unalias(u)
			continue
		case *types.Named:
			return u.Origin().Obj()
		}
		return nil
	}
}

// ---------- one file set ----------

type loaded struct {
	fset   *token.FileSet
	pkgs   []*types.Package
	infos  map[*types.Package]*types.Info
	files  map[*types.Package][]*ast.File
	srcs   map[string][]byte // repo-relative path -> source
	inTree map[string]bool   // import path -> under internal/sync
}

func load(env []string) *loaded {
	out := goCmd(env, "list", "-json", "-tags", tags, "./...")
	dec := json.NewDecoder(bytes.NewReader(out))
	var all []goPkg
	for {
		var p goPkg
		if err := dec.Decode(&p); err == io.EOF {
			break
		} else if err != nil {
			fail("go list: %v", err)
		}
		all = append(all, p)
	}
	module := ""
	for _, p := range all {
		if strings.HasSuffix(p.ImportPath, "/"+syncRel) {
			module = strings.TrimSuffix(p.ImportPath, "/"+syncRel)
		}
	}
	syncPrefix := module + "/" + syncRel
	isTree := func(path string) bool { return path == syncPrefix || strings.HasPrefix(path, syncPrefix+"/") }
	depOut := goCmd(env, "list", "-deps", "-test", "-tags", tags, "-f", "{{.ImportPath}}", "./...")
	var deps []string
	seen := map[string]bool{}
	for _, l := range strings.Split(string(depOut), "\n") {
		l = strings.TrimSpace(l)
		if l == "" || strings.Contains(l, " ") || strings.HasSuffix(l, ".test") || l == "unsafe" || l == "C" || seen[l] {
			continue
		}
		seen[l] = true
		deps = append(deps, l)
	}
	sort.Strings(deps)
	expOut := goCmd(env, append([]string{"list", "-export", "-tags", tags, "-f", "{{.ImportPath}}\t{{.Export}}"}, deps...)...)
	exports := map[string]string{}
	for _, l := range strings.Split(string(expOut), "\n") {
		parts := strings.Split(l, "\t")
		if len(parts) == 2 && parts[1] != "" {
			exports[parts[0]] = parts[1]
		}
	}
	L := &loaded{
		fset: token.NewFileSet(), infos: map[*types.Package]*types.Info{}, files: map[*types.Package][]*ast.File{},
		srcs: map[string][]byte{}, inTree: map[string]bool{},
	}
	imp := importer.ForCompiler(L.fset, "gc", func(path string) (io.ReadCloser, error) {
		f, ok := exports[path]
		if !ok {
			return nil, fmt.Errorf("no export data for %s", path)
		}
		return os.Open(f)
	})
	for _, p := range all {
		uses := isTree(p.ImportPath)
		for _, i := range append(append([]string{}, p.Imports...), p.TestImports...) {
			if isTree(i) {
				uses = true
			}
		}
		if !uses {
			continue
		}
		if len(p.XTestGoFiles) > 0 {
			fail("%s: external test packages are not handled", p.ImportPath)
		}
		var files []*ast.File
		for _, n := range append(append([]string{}, p.GoFiles...), p.TestGoFiles...) {
			path := filepath.Join(p.Dir, n)
			b, err := os.ReadFile(path)
			if err != nil {
				fail("%v", err)
			}
			f, err := parser.ParseFile(L.fset, path, b, parser.ParseComments)
			if err != nil {
				fail("parse %s: %v", path, err)
			}
			rel, _ := filepath.Rel(*repoFlag, path)
			L.srcs[rel] = b
			files = append(files, f)
		}
		info := &types.Info{Uses: map[*ast.Ident]types.Object{}, Defs: map[*ast.Ident]types.Object{}, Scopes: map[ast.Node]*types.Scope{}}
		var errs []string
		conf := types.Config{Importer: imp, Error: func(err error) { errs = append(errs, err.Error()) }}
		pkg, _ := conf.Check(p.ImportPath, L.fset, files, info)
		if len(errs) > 0 {
			fail("type-check %s: %s", p.ImportPath, errs[0])
		}
		L.pkgs = append(L.pkgs, pkg)
		L.infos[pkg] = info
		L.files[pkg] = files
		L.inTree[p.ImportPath] = isTree(p.ImportPath)
	}
	return L
}

// key names an object of a tree package independently of the load:
// package path, owner type ("" for package level) and name.
type key struct{ pkg, owner, name string }

func keyOf(obj types.Object) (key, bool) {
	if obj == nil || obj.Pkg() == nil {
		return key{}, false
	}
	switch o := obj.(type) {
	case *types.Func:
		o = o.Origin()
		sig := o.Type().(*types.Signature)
		if sig.Recv() == nil {
			if o.Parent() == o.Pkg().Scope() {
				return key{o.Pkg().Path(), "", o.Name()}, true
			}
			return key{}, false
		}
		tn := recvNamed(sig.Recv().Type())
		if tn == nil || tn.Parent() != tn.Pkg().Scope() {
			return key{}, false
		}
		return key{o.Pkg().Path(), tn.Name(), o.Name()}, true
	case *types.Var:
		if o.IsField() {
			return key{}, false // fields are keyed by fieldOwner
		}
	}
	if obj.Parent() == obj.Pkg().Scope() {
		return key{obj.Pkg().Path(), "", obj.Name()}, true
	}
	return key{}, false
}

// fieldOwners maps each field of a named struct type of pkg to its key.
func fieldOwners(pkg *types.Package, into map[*types.Var]key) {
	for _, n := range pkg.Scope().Names() {
		tn, ok := pkg.Scope().Lookup(n).(*types.TypeName)
		if !ok {
			continue
		}
		if st, ok := tn.Type().Underlying().(*types.Struct); ok {
			for i := 0; i < st.NumFields(); i++ {
				f := st.Field(i)
				into[f] = key{pkg.Path(), tn.Name(), f.Name()}
			}
		}
	}
}

type state struct {
	base       map[[2]string]bool
	candidates map[key]string // -> kind
	crossing   map[key]bool
	kept       map[key]string // reason
	ifaceNames map[string]bool
	literals   map[string]bool // identifiers named in string literals
}

func (s *state) analyse(L *loaded) {
	fieldKey := map[*types.Var]key{}
	for _, pkg := range L.pkgs {
		fieldOwners(pkg, fieldKey)
	}
	resolve := func(obj types.Object) (key, string, bool) {
		if v, ok := obj.(*types.Var); ok && v.IsField() {
			k, found := fieldKey[v.Origin()]
			if !found {
				// a field of a type loaded from export data
				return key{}, "", false
			}
			return k, "field", true
		}
		k, ok := keyOf(obj)
		if !ok {
			return key{}, "", false
		}
		if k.owner != "" {
			return k, "method", true
		}
		return k, "decl", true
	}
	treePath := func(path string) bool { return strings.Contains(path+"/", "/"+syncRel+"/") }
	// crossing: a use from another package. Objects loaded from export data
	// are other objects than the source ones, so keys compare them.
	var crossObjs []types.Object
	for _, pkg := range L.pkgs {
		info := L.infos[pkg]
		for _, obj := range info.Uses {
			if obj.Pkg() == nil || obj.Pkg().Path() == pkg.Path() || !treePath(obj.Pkg().Path()) {
				continue
			}
			k, ok := keyOf(obj)
			if v, isVar := obj.(*types.Var); isVar && v.IsField() {
				tn := ownerOfField(v)
				if tn == nil {
					continue
				}
				k, ok = key{obj.Pkg().Path(), tn.Name(), v.Name()}, true
			}
			if ok {
				s.crossing[k] = true
				crossObjs = append(crossObjs, obj)
			}
		}
		// interface method names anywhere loaded
		for _, tv := range pkg.Scope().Names() {
			if tn, ok := pkg.Scope().Lookup(tv).(*types.TypeName); ok {
				if it, ok := tn.Type().Underlying().(*types.Interface); ok {
					for i := 0; i < it.NumMethods(); i++ {
						s.ifaceNames[it.Method(i).Name()] = true
					}
				}
			}
		}
	}
	// types reachable from what crosses
	seen := map[types.Type]bool{}
	var walk func(t types.Type)
	walk = func(t types.Type) {
		if t == nil || seen[t] {
			return
		}
		seen[t] = true
		switch u := t.(type) {
		case *types.Named:
			obj := u.Origin().Obj()
			if obj.Pkg() != nil && treePath(obj.Pkg().Path()) {
				s.crossing[key{obj.Pkg().Path(), "", obj.Name()}] = true
			}
			for i := 0; i < u.TypeArgs().Len(); i++ {
				walk(u.TypeArgs().At(i))
			}
			walk(u.Underlying())
		case *types.Alias:
			walk(types.Unalias(u))
		case *types.Pointer:
			walk(u.Elem())
		case *types.Slice:
			walk(u.Elem())
		case *types.Array:
			walk(u.Elem())
		case *types.Map:
			walk(u.Key())
			walk(u.Elem())
		case *types.Chan:
			walk(u.Elem())
		case *types.Signature:
			walk(u.Params())
			walk(u.Results())
		case *types.Tuple:
			for i := 0; i < u.Len(); i++ {
				walk(u.At(i).Type())
			}
		case *types.Struct:
			for i := 0; i < u.NumFields(); i++ {
				if u.Field(i).Exported() {
					walk(u.Field(i).Type())
				}
			}
		case *types.Interface:
			for i := 0; i < u.NumMethods(); i++ {
				walk(u.Method(i).Type())
			}
		}
	}
	for _, obj := range crossObjs {
		walk(obj.Type())
	}
	// candidates: exported now, unexported at base, declared in the tree
	for _, pkg := range L.pkgs {
		if !L.inTree[pkg.Path()] {
			continue
		}
		info := L.infos[pkg]
		for id, obj := range info.Defs {
			if obj == nil || !obj.Exported() {
				continue
			}
			k, kind, ok := resolve(obj)
			if !ok || k.pkg != pkg.Path() {
				continue
			}
			_ = id
			owner := strings.ToLower(k.owner)
			if !s.base[[2]string{owner, lowerFirst(k.name)}] || s.base[[2]string{owner, k.name}] {
				continue
			}
			if kind == "method" {
				if tn, ok := pkg.Scope().Lookup(k.owner).(*types.TypeName); ok {
					if _, isIface := tn.Type().Underlying().(*types.Interface); isIface {
						kind = "interface method"
					}
				}
			}
			s.candidates[k] = kind
		}
	}
}

func ownerOfField(v *types.Var) *types.TypeName {
	pkg := v.Pkg()
	for _, n := range pkg.Scope().Names() {
		tn, ok := pkg.Scope().Lookup(n).(*types.TypeName)
		if !ok {
			continue
		}
		if st, ok := tn.Type().Underlying().(*types.Struct); ok {
			for i := 0; i < st.NumFields(); i++ {
				if st.Field(i) == v.Origin() {
					return tn
				}
			}
		}
	}
	return nil
}

type edit struct {
	off, end int
	text     string
}

func apply(src []byte, edits []edit) []byte {
	sort.Slice(edits, func(i, j int) bool { return edits[i].off < edits[j].off })
	var b bytes.Buffer
	last := 0
	for i, e := range edits {
		if i > 0 && e == edits[i-1] {
			continue
		}
		if e.off < last {
			fail("overlapping edits at %d", e.off)
		}
		b.Write(src[last:e.off])
		b.WriteString(e.text)
		last = e.end
	}
	b.Write(src[last:])
	return b.Bytes()
}

// renames returns, per repo-relative file, the edits that rename the chosen
// keys, after the scope checks; a key whose new name would be captured is
// moved to kept.
func (s *state) renames(L *loaded, chosen map[key]bool) map[string][]edit {
	fieldKey := map[*types.Var]key{}
	for _, pkg := range L.pkgs {
		fieldOwners(pkg, fieldKey)
	}
	type site struct {
		file  string
		id    *ast.Ident
		pkg   *types.Package
		isDef bool
	}
	sites := map[key][]site{}
	for _, pkg := range L.pkgs {
		if !L.inTree[pkg.Path()] {
			continue
		}
		info := L.infos[pkg]
		for _, f := range L.files[pkg] {
			rel, _ := filepath.Rel(*repoFlag, L.fset.Position(f.Pos()).Filename)
			ast.Inspect(f, func(n ast.Node) bool {
				id, ok := n.(*ast.Ident)
				if !ok {
					return true
				}
				obj, isDef := info.Defs[id], true
				if u := info.Uses[id]; u != nil {
					obj, isDef = u, false
				}
				if obj == nil {
					return true
				}
				var k key
				var found bool
				if v, isVar := obj.(*types.Var); isVar && v.IsField() {
					if v.Embedded() && !isDef {
						return true // a promoted field takes its type's name
					}
					k, found = fieldKey[v.Origin()]
				} else {
					k, found = keyOf(obj)
				}
				if found && chosen[k] {
					sites[k] = append(sites[k], site{rel, id, pkg, isDef})
				}
				return true
			})
		}
	}
	edits := map[string][]edit{}
	for k := range chosen {
		low := lowerFirst(k.name)
		bad := ""
		for _, st := range sites[k] {
			if k.owner == "" {
				scope := st.pkg.Scope().Innermost(st.id.Pos())
				if scope == nil {
					scope = st.pkg.Scope()
				}
				if _, o := scope.LookupParent(low, st.id.Pos()); o != nil {
					bad = fmt.Sprintf("%s resolves to %s at %s", low, o, L.fset.Position(st.id.Pos()))
					break
				}
				// a later local declaration of the name in a scope that holds the use
				if o := st.pkg.Scope().Lookup(low); o != nil {
					bad = low + " exists in the package"
					break
				}
			}
		}
		if k.owner != "" {
			pkg := pkgByPath(L, k.pkg)
			if tn, ok := pkg.Scope().Lookup(k.owner).(*types.TypeName); ok {
				if o, _, _ := types.LookupFieldOrMethod(tn.Type(), true, pkg, low); o != nil {
					bad = k.owner + "." + low + " exists"
				}
			}
		}
		if bad != "" {
			s.kept[k] = "name capture: " + bad
			continue
		}
		for _, st := range sites[k] {
			o := L.fset.Position(st.id.Pos()).Offset
			edits[st.file] = append(edits[st.file], edit{o, o + len(st.id.Name), low})
		}
	}
	// doc comments whose first word is the renamed declaration
	for _, pkg := range L.pkgs {
		if !L.inTree[pkg.Path()] {
			continue
		}
		info := L.infos[pkg]
		for _, f := range L.files[pkg] {
			rel, _ := filepath.Rel(*repoFlag, L.fset.Position(f.Pos()).Filename)
			ast.Inspect(f, func(n ast.Node) bool {
				var doc *ast.CommentGroup
				var name *ast.Ident
				switch x := n.(type) {
				case *ast.FuncDecl:
					doc, name = x.Doc, x.Name
				case *ast.TypeSpec:
					doc, name = x.Doc, x.Name
				case *ast.ValueSpec:
					if len(x.Names) == 1 {
						doc, name = x.Doc, x.Names[0]
					}
				case *ast.Field:
					if len(x.Names) == 1 {
						doc, name = x.Doc, x.Names[0]
					}
				case *ast.GenDecl:
					if !x.Lparen.IsValid() && len(x.Specs) == 1 {
						switch sp := x.Specs[0].(type) {
						case *ast.TypeSpec:
							doc, name = x.Doc, sp.Name
						case *ast.ValueSpec:
							if len(sp.Names) == 1 {
								doc, name = x.Doc, sp.Names[0]
							}
						}
					}
				}
				if doc == nil || name == nil {
					return true
				}
				obj := info.Defs[name]
				var k key
				var ok bool
				if v, isVar := obj.(*types.Var); isVar && v.IsField() {
					k, ok = fieldKey[v.Origin()]
				} else {
					k, ok = keyOf(obj)
				}
				if !ok || !chosen[k] || s.kept[k] != "" {
					return true
				}
				c := doc.List[0]
				if strings.HasPrefix(c.Text, "// "+name.Name+" ") || strings.HasPrefix(c.Text, "// "+name.Name+"\n") || c.Text == "// "+name.Name {
					o := L.fset.Position(c.Pos()).Offset + 3
					e := edit{o, o + len(name.Name), lowerFirst(name.Name)}
					for _, x := range edits[rel] {
						if x == e {
							return true
						}
					}
					edits[rel] = append(edits[rel], e)
				}
				return true
			})
		}
	}
	return edits
}

func pkgByPath(L *loaded, path string) *types.Package {
	for _, p := range L.pkgs {
		if p.Path() == path {
			return p
		}
	}
	fail("package %s not loaded", path)
	return nil
}

func shortPkg(path string) string { return path[strings.Index(path, syncRel):] }

var identRE = regexp.MustCompile(`[A-Za-z_][A-Za-z0-9_]*`)

func main() {
	flag.Parse()
	abs, err := filepath.Abs(*repoFlag)
	if err != nil {
		fail("%v", err)
	}
	*repoFlag = abs
	s := &state{
		base: baseNames(), candidates: map[key]string{}, crossing: map[key]bool{}, kept: map[key]string{},
		ifaceNames: map[string]bool{}, literals: map[string]bool{},
	}
	sets := []*loaded{
		load([]string{"GOOS=linux", "GOARCH=amd64"}),
		load([]string{"GOOS=darwin", "GOARCH=arm64", "CGO_ENABLED=0"}),
	}
	for _, L := range sets {
		s.analyse(L)
		for _, src := range L.srcs {
			fset := token.NewFileSet()
			f, err := parser.ParseFile(fset, "", src, parser.SkipObjectResolution)
			if err != nil {
				continue
			}
			ast.Inspect(f, func(n ast.Node) bool {
				if lit, ok := n.(*ast.BasicLit); ok && lit.Kind == token.STRING {
					v, err := strconv.Unquote(lit.Value)
					if err != nil {
						v = lit.Value
					}
					for _, w := range identRE.FindAllString(v, -1) {
						s.literals[w] = true
					}
				}
				return true
			})
		}
	}
	chosen := map[key]bool{}
	for k, kind := range s.candidates {
		switch {
		case s.crossing[k]:
		case kind == "interface method":
			s.kept[k] = "an interface method"
		case kind == "method" && s.ifaceNames[k.name]:
			s.kept[k] = "an interface of the module declares a method " + k.name
		case s.literals[k.name]:
			s.kept[k] = "a string literal names " + k.name
		default:
			chosen[k] = true
		}
	}
	perSet := make([]map[string][]edit, len(sets))
	for i, L := range sets {
		perSet[i] = s.renames(L, chosen)
	}
	for k := range s.kept {
		delete(chosen, k)
	}
	if len(s.kept) > 0 {
		// a key kept by the second set's scope check was renamed in the first: redo both
		for i, L := range sets {
			perSet[i] = s.renames(L, chosen)
		}
	}
	// one rewrite per file
	files := map[string][]byte{}
	for i, L := range sets {
		for rel, src := range L.srcs {
			out := apply(append([]byte{}, src...), perSet[i][rel])
			if len(perSet[i][rel]) == 0 {
				continue
			}
			formatted, err := format.Source(out)
			if err != nil {
				fail("format %s: %v", rel, err)
			}
			if prev, ok := files[rel]; ok && !bytes.Equal(prev, formatted) {
				fail("the Linux and macOS rewrites of %s differ", rel)
			}
			files[rel] = formatted
		}
	}
	// the docs: backticked mentions, when no identifier that stays exported shares the name
	staying := map[string]bool{}
	for _, L := range sets {
		for _, pkg := range L.pkgs {
			info := L.infos[pkg]
			for _, obj := range info.Defs {
				if obj != nil && obj.Exported() {
					k, ok := keyOf(obj)
					if !ok || !chosen[k] {
						staying[obj.Name()] = true
					}
				}
			}
		}
	}
	docNames := map[string]string{}
	var report []string
	for k := range chosen {
		report = append(report, fmt.Sprintf("unexport %s %s %s", shortPkg(k.pkg), k.owner, k.name))
		if !staying[k.name] {
			docNames[k.name] = lowerFirst(k.name)
			if k.owner != "" {
				docNames[k.owner+"."+k.name] = k.owner + "." + lowerFirst(k.name)
			}
		}
	}
	for k, why := range s.kept {
		report = append(report, fmt.Sprintf("keep %s %s %s: %s", shortPkg(k.pkg), k.owner, k.name, why))
	}
	sort.Strings(report)
	nCross := 0
	for k := range s.candidates {
		if s.crossing[k] {
			nCross++
		}
	}
	fmt.Printf("%d candidates: %d named by another package, %d kept, %d un-exported; %d Go files change\n",
		len(s.candidates), nCross, len(s.kept), len(chosen), len(files))
	for _, l := range report {
		fmt.Println(l)
	}
	if *dryFlag {
		return
	}
	for rel, b := range files {
		if err := os.WriteFile(filepath.Join(*repoFlag, rel), b, 0o644); err != nil {
			fail("%v", err)
		}
	}
	tick := regexp.MustCompile("`([A-Za-z_][A-Za-z0-9_.]*)`")
	for _, dir := range []string{"spec/design", "spec/reference"} {
		docs, _ := filepath.Glob(filepath.Join(*repoFlag, dir, "*.md"))
		for _, d := range docs {
			if strings.HasSuffix(d, "retired-backlog-ids.md") || strings.HasSuffix(d, "live-incidents.md") {
				continue
			}
			b, err := os.ReadFile(d)
			if err != nil {
				fail("%v", err)
			}
			nb := tick.ReplaceAllFunc(b, func(m []byte) []byte {
				inner := string(m[1 : len(m)-1])
				if r, ok := docNames[inner]; ok {
					return []byte("`" + r + "`")
				}
				return m
			})
			if !bytes.Equal(b, nb) {
				if err := os.WriteFile(d, nb, 0o644); err != nil {
					fail("%v", err)
				}
			}
		}
	}
}
