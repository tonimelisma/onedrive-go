// splitmove moves the files of internal/sync into the packages the end-state
// simulator (notes/onedrive-go-sync-split-2026-11-10/tools/endstate) assigns
// them, and rewrites every reference so the tree compiles unchanged in
// behaviour.
//
//	go -C <repo> run /path/splitmove.go -repo <repo> -sim <simulator out dir> -conf <final.txt>
//
// What it does, and nothing else:
//
//   - every file of internal/sync goes to its package's directory; its
//     package clause names that package. A file's package is the package
//     the simulator gives all of its declarations (rawdecls.tsv, tests.tsv);
//     a file whose declarations disagree stops the run.
//   - the test-helper packages' files (fixture_*_test.go, storetest_*_test.go,
//     race_*_test.go) become ordinary files: the prefix and "_test" go.
//   - a reference to a package-level identifier that now lives in another
//     package is qualified with that package's name, and the file imports it.
//   - an unexported identifier of a test-helper package (declaration, field or
//     method) that another package uses is capitalised everywhere, with the
//     first word of its doc comment.
//   - the private test copies (testcopy lines of -conf) are written, rendered
//     for the copying package, into helper_copies_test.go of that package.
//   - in every other package that imports internal/sync, a qualifier
//     naming an identifier that moved names the new package; the import list
//     follows.
//   - testdata directories named by a "testdata/<dir>/" string in the test
//     files of one package move to that package's directory.
//
// Every file it writes is gofmt-formatted (go/format). Type information comes
// from go/types over the export data `go list -export` produces, for the
// Linux and the macOS file sets; a file both sets type-check must get the
// same rewrite from each. Standard library only.

func fail(format string, a ...any) {
	fmt.Fprintf(os.Stderr, "splitmove: "+format+"\n", a...)
	os.Exit(1)
}

// ---------- inputs ----------

type plan struct {
	declHome map[string]string          // declaration id (name, or Type.method) -> package
	fileHome map[string]string          // file base name -> package (from the simulator)
	copies   map[string]map[string]bool // package -> declaration ids its tests keep a private copy of
	helper   map[string]bool            // test-helper packages
}

func readTSV(path string) [][]string {
	b, err := os.ReadFile(path)
	if err != nil {
		fail("%v", err)
	}
	var rows [][]string
	for i, l := range strings.Split(strings.TrimRight(string(b), "\n"), "\n") {
		if i == 0 || l == "" {
			continue
		}
		rows = append(rows, strings.Split(l, "\t"))
	}
	return rows
}

func loadPlan() *plan {
	p := &plan{declHome: map[string]string{}, fileHome: map[string]string{}, copies: map[string]map[string]bool{}, helper: map[string]bool{}}
	setFile := func(file, pkg string) {
		if old, ok := p.fileHome[file]; ok && old != pkg {
			fail("file %s has declarations in %s and %s", file, old, pkg)
		}
		p.fileHome[file] = pkg
	}
	// rawdecls: id kind recv file test loc typeof pkg how
	for _, r := range readTSV(filepath.Join(*simFlag, "rawdecls.tsv")) {
		if r[4] != "false" {
			continue
		}
		p.declHome[r[0]] = r[7]
		if !strings.HasPrefix(r[0], "_@") {
			setFile(r[3], r[7])
		}
	}
	// testdecls: decl kind file loc pkg ...
	for _, r := range readTSV(filepath.Join(*simFlag, "testdecls.tsv")) {
		if !strings.HasPrefix(r[0], "_@") {
			p.declHome[r[0]] = r[4]
		}
	}
	// tests: test_file home loc_by_pkg ...; loc_by_pkg names the one package the file's declarations go to
	for _, r := range readTSV(filepath.Join(*simFlag, "tests.tsv")) {
		parts := strings.Fields(r[2])
		if len(parts) != 1 {
			fail("test file %s spans packages: %q", r[0], r[2])
		}
		setFile(r[0], strings.SplitN(parts[0], "=", 2)[0])
	}
	b, err := os.ReadFile(*confFlag)
	if err != nil {
		fail("%v", err)
	}
	for _, l := range strings.Split(string(b), "\n") {
		f := strings.Fields(strings.SplitN(l, "#", 2)[0])
		switch {
		case len(f) >= 2 && f[0] == "testcopy":
			if p.copies[f[1]] == nil {
				p.copies[f[1]] = map[string]bool{}
			}
			for _, d := range f[2:] {
				p.copies[f[1]][d] = true
			}
		case len(f) >= 2 && f[0] == "helper":
			for _, h := range f[1:] {
				p.helper[h] = true
			}
		}
	}
	for h := range p.helper {
		if helperPrefix[h] == "" {
			fail("helper package %s has no file prefix", h)
		}
	}
	return p
}

// ---------- go list ----------

type goPkg struct {
	Dir, ImportPath, Name                        string
	GoFiles, TestGoFiles, XTestGoFiles, CgoFiles []string
	Imports, TestImports, XTestImports           []string
}

func goCmd(env []string, args ...string) []byte {
	cmd := exec.Command("go", args...)
	cmd.Dir = *repoFlag
	cmd.Env = append(os.Environ(), env...)
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	out, err := cmd.Output()
	if err != nil {
		fail("go %s: %v\n%s", strings.Join(args, " "), err, stderr.String())
	}
	return out
}

func goList(env []string) []goPkg {
	out := goCmd(env, "list", "-json", "-tags", tags, "./...")
	dec := json.NewDecoder(bytes.NewReader(out))
	var pkgs []goPkg
	for {
		var p goPkg
		if err := dec.Decode(&p); err == io.EOF {
			break
		} else if err != nil {
			fail("go list: %v", err)
		}
		pkgs = append(pkgs, p)
	}
	return pkgs
}

func exportData(env []string) map[string]string {
	depOut := goCmd(env, "list", "-deps", "-test", "-tags", tags, "-f", "{{.ImportPath}}", "./...")
	var deps []string
	seen := map[string]bool{}
	for _, l := range strings.Split(string(depOut), "\n") {
		l = strings.TrimSpace(l)
		if l == "" || strings.Contains(l, " ") || strings.HasSuffix(l, ".test") || l == "unsafe" || l == "C" || seen[l] {
			continue
		}
		seen[l] = true
		deps = append(deps, l)
	}
	sort.Strings(deps)
	out := goCmd(env, append([]string{"list", "-export", "-tags", tags, "-f", "{{.ImportPath}}\t{{.Export}}"}, deps...)...)
	m := map[string]string{}
	for _, l := range strings.Split(string(out), "\n") {
		parts := strings.Split(l, "\t")
		if len(parts) == 2 && parts[1] != "" {
			m[parts[0]] = parts[1]
		}
	}
	return m
}

// ---------- edits ----------

type edit struct {
	off, end int
	text     string
}

func apply(src []byte, edits []edit) []byte {
	sort.Slice(edits, func(i, j int) bool {
		if edits[i].off != edits[j].off {
			return edits[i].off < edits[j].off
		}
		return edits[i].end < edits[j].end
	})
	var b bytes.Buffer
	last := 0
	for _, e := range edits {
		if e.off < last {
			fail("overlapping edits at offset %d", e.off)
		}
		b.Write(src[last:e.off])
		b.WriteString(e.text)
		last = e.end
	}
	b.Write(src[last:])
	return b.Bytes()
}

func exported(name string) string {
	r, n := utf8.DecodeRuneInString(name)
	return string(unicode.ToUpper(r)) + name[n:]
}

// ---------- analysis of one file set ----------

// result is what one GOOS's type-check decides for each file.
type result struct {
	files  map[string][]byte // repo-relative destination path -> formatted content
	source map[string]string // destination -> repo-relative source path
	marked map[string]bool   // helper identifiers capitalised (keys as in ownerKey)
	errors []string
}

type analysis struct {
	p        *plan
	fset     *token.FileSet
	pkg      *types.Package
	info     *types.Info
	syncPath string
	module   string
	files    map[string]*ast.File // base name -> file
	srcs     map[string][]byte    // base name -> source
	fieldKey map[*types.Var]string
	marked   map[string]bool
	newName  map[string]string // marked key -> its exported name
	declFile map[string]string // declaration id -> base name of its file
	fileHome func(base string) string
	errors   map[string]bool
}

func (a *analysis) fileOf(pos token.Pos) string { return filepath.Base(a.fset.Position(pos).Filename) }
func (a *analysis) off(pos token.Pos) int       { return a.fset.Position(pos).Offset }

// owner returns the declaration id an object belongs to (a package-level
// name, or the named type of a method or field) and the key the object is
// marked under; ok is false for objects of no package-level declaration.
func (a *analysis) owner(obj types.Object) (ownerID, key string, ok bool) {
	if obj == nil || obj.Pkg() != a.pkg {
		return "", "", false
	}
	switch o := obj.(type) {
	case *types.Func:
		o = o.Origin()
		sig := o.Type().(*types.Signature)
		if sig.Recv() == nil {
			if o.Parent() == a.pkg.Scope() {
				return o.Name(), o.Name(), true
			}
			return "", "", false
		}
		tn := recvNamed(sig.Recv().Type())
		if tn == nil || tn.Parent() != a.pkg.Scope() {
			return "", "", false
		}
		return tn.Name(), tn.Name() + "." + o.Name(), true
	case *types.Var:
		if o.IsField() {
			k, found := a.fieldKey[o.Origin()]
			if !found {
				return "", "", false
			}
			return strings.SplitN(k, ".", 2)[0], k, true
		}
	}
	if obj.Parent() == a.pkg.Scope() {
		return obj.Name(), obj.Name(), true
	}
	return "", "", false
}

func isSig(t types.Type) bool  { _, ok := t.Underlying().(*types.Signature); return ok }
func isChan(t types.Type) bool { _, ok := t.Underlying().(*types.Chan); return ok }

func recvNamed(t types.Type) *types.TypeName {
	for {
		switch u := t.(type) {
		case *types.Pointer:
			t = u.Elem()
			continue
		case *types.Alias:
			t = types.Unalias(u)
			continue
		case *types.Named:
			return u.Origin().Obj()
		}
		return nil
	}
}

func (a *analysis) buildFieldKeys() {
	a.fieldKey = map[*types.Var]string{}
	var walk func(prefix string, t types.Type)
	walk = func(prefix string, t types.Type) {
		st, ok := t.(*types.Struct)
		if !ok {
			return
		}
		for i := 0; i < st.NumFields(); i++ {
			f := st.Field(i)
			k := prefix + "." + f.Name()
			if _, dup := a.fieldKey[f]; !dup {
				a.fieldKey[f] = k
			}
			walk(k, f.Type())
		}
	}
	for _, name := range a.pkg.Scope().Names() {
		if tn, ok := a.pkg.Scope().Lookup(name).(*types.TypeName); ok {
			walk(name, tn.Type().Underlying())
		}
	}
}

// home is where declaration id lives as seen from code in package loc: a
// private copy of loc's own, or the declaration's package.
func (a *analysis) home(id, loc string) string {
	if a.p.copies[loc][id] {
		return loc
	}
	if h, ok := a.p.declHome[id]; ok {
		return h
	}
	// a method of a test type lives with its receiver; a constant of a group with its file
	if i := strings.Index(id, "."); i > 0 {
		return a.home(id[:i], loc)
	}
	if f, ok := a.declFile[id]; ok {
		return a.fileHome(f)
	}
	a.errors[fmt.Sprintf("no package for declaration %s", id)] = true
	return loc
}

// embeddedType is the type name an embedded field takes its name from.
func embeddedType(v *types.Var) *types.TypeName {
	if !v.Embedded() {
		return nil
	}
	return recvNamed(v.Type())
}

// declRange is one top-level declaration's source range in a file.
type declRange struct {
	id         string
	start, end token.Pos
	keyword    string // for a spec cut out of a parenthesized group: "var ", "const ", "type "; else ""
	doc        *ast.CommentGroup
	name       *ast.Ident
	group      *ast.GenDecl // the parenthesized group a spec is cut out of, nil otherwise
}

// directives are the //nolint and //go: lines of a comment group: what a
// declaration cut out of its group must carry with it.
func directives(doc *ast.CommentGroup) []string {
	if doc == nil {
		return nil
	}
	var out []string
	for _, c := range doc.List {
		if strings.HasPrefix(c.Text, "//nolint") || strings.HasPrefix(c.Text, "//go:") {
			out = append(out, c.Text)
		}
	}
	return out
}

func (a *analysis) declRanges(f *ast.File) []declRange {
	var out []declRange
	for _, d := range f.Decls {
		switch x := d.(type) {
		case *ast.FuncDecl:
			start := x.Pos()
			if x.Doc != nil {
				start = x.Doc.Pos()
			}
			id := x.Name.Name
			if x.Recv != nil && len(x.Recv.List) > 0 {
				id = recvBase(x.Recv.List[0].Type) + "." + id
			}
			out = append(out, declRange{id: id, start: start, end: x.End(), doc: x.Doc, name: x.Name})
		case *ast.GenDecl:
			if x.Tok == token.IMPORT {
				continue
			}
			for _, sp := range x.Specs {
				var names []*ast.Ident
				var doc, line *ast.CommentGroup
				switch s := sp.(type) {
				case *ast.TypeSpec:
					names, doc, line = []*ast.Ident{s.Name}, s.Doc, s.Comment
				case *ast.ValueSpec:
					names, doc, line = s.Names, s.Doc, s.Comment
				}
				dr := declRange{start: sp.Pos(), end: sp.End(), doc: doc}
				if line != nil {
					// a trailing comment, a directive among them, goes with its spec
					dr.end = line.End()
				}
				if x.Lparen.IsValid() {
					dr.keyword = x.Tok.String() + " "
					dr.group = x
					if doc != nil {
						dr.start = doc.Pos()
					}
				} else {
					dr.start, dr.end, dr.doc = x.Pos(), x.End(), x.Doc
					if x.Doc != nil {
						dr.start = x.Doc.Pos()
					}
					if line != nil && line.End() > dr.end {
						dr.end = line.End()
					}
				}
				for _, n := range names {
					r := dr
					r.id, r.name = n.Name, n
					out = append(out, r)
				}
			}
		}
	}
	return out
}

func recvBase(e ast.Expr) string {
	for {
		switch x := e.(type) {
		case *ast.StarExpr:
			e = x.X
		case *ast.ParenExpr:
			e = x.X
		case *ast.IndexExpr:
			e = x.X
		case *ast.IndexListExpr:
			e = x.X
		case *ast.Ident:
			return x.Name
		default:
			return ""
		}
	}
}

// visit calls fn for every identifier in [start, end) of file f that names an
// object of the sync package, with whether it is a use (else a definition).
func (a *analysis) visit(f *ast.File, start, end token.Pos, fn func(id *ast.Ident, obj types.Object, use bool)) {
	ast.Inspect(f, func(n ast.Node) bool {
		if n == nil {
			return false
		}
		if n.End() <= start || n.Pos() >= end {
			return false
		}
		id, ok := n.(*ast.Ident)
		if !ok {
			return true
		}
		if obj := a.info.Uses[id]; obj != nil {
			fn(id, obj, true)
			return true
		}
		if obj := a.info.Defs[id]; obj != nil {
			fn(id, obj, false)
		}
		return true
	})
}

// mark records which helper identifiers have to be exported: an unexported
// object of a helper package that code in another package names.
func (a *analysis) mark(f *ast.File, start, end token.Pos, loc string) {
	a.visit(f, start, end, func(id *ast.Ident, obj types.Object, use bool) {
		if !use {
			return
		}
		if v, ok := obj.(*types.Var); ok && v.IsField() {
			if tn := embeddedType(v); tn != nil && tn.Pkg() == a.pkg {
				obj = tn
			}
		}
		ownerID, key, ok := a.owner(obj)
		if !ok || token.IsExported(obj.Name()) {
			return
		}
		h := a.home(ownerID, loc)
		if h == loc {
			return
		}
		if !a.p.helper[h] {
			a.errors[fmt.Sprintf("%s: unexported %s (package %s) named from package %s", a.fset.Position(id.Pos()), key, h, loc)] = true
			return
		}
		a.marked[key] = true
	})
}

// render returns the edits that make [start, end) of file f correct code of
// package loc, and the packages that code then names.
func (a *analysis) render(f *ast.File, start, end token.Pos, loc string) ([]edit, map[string]bool) {
	var edits []edit
	need := map[string]bool{}
	// a capitalised helper identifier keeps its old name only inside a private copy
	nameFor := func(obj types.Object, ownerID, key string) string {
		if a.marked[key] && !a.p.copies[loc][ownerID] {
			return a.newName[key]
		}
		return obj.Name()
	}
	a.visit(f, start, end, func(id *ast.Ident, obj types.Object, use bool) {
		if !use && a.info.Uses[id] != nil {
			return
		}
		var text string
		var embedded *types.TypeName
		if v, ok := obj.(*types.Var); ok && v.IsField() {
			if tn := embeddedType(v); tn != nil && tn.Pkg() == a.pkg {
				embedded = tn
			}
		}
		if embedded != nil {
			// a promoted field takes the name of its type
			ownerID, key, ok := a.owner(embedded)
			if !ok {
				return
			}
			text = nameFor(embedded, ownerID, key)
		} else {
			ownerID, key, ok := a.owner(obj)
			if !ok {
				return
			}
			text = nameFor(obj, ownerID, key)
			if use && ownerID == key { // a package-level identifier
				if h := a.home(ownerID, loc); h != loc {
					text = h + "." + text
					need[h] = true
				}
			}
		}
		if text != id.Name {
			edits = append(edits, edit{a.off(id.Pos()), a.off(id.End()), text})
		}
	})
	return edits, need
}

// docRename renames the first word of a capitalised declaration's doc comment.
func (a *analysis) docRename(dr declRange, loc string) []edit {
	if dr.doc == nil || dr.name == nil {
		return nil
	}
	obj := a.info.Defs[dr.name]
	ownerID, key, ok := a.owner(obj)
	if !ok || !a.marked[key] || a.p.copies[loc][ownerID] {
		return nil
	}
	var edits []edit
	for _, c := range dr.doc.List {
		text := c.Text
		if !strings.HasPrefix(text, "// "+dr.name.Name+" ") {
			continue
		}
		o := a.off(c.Pos()) + 3
		edits = append(edits, edit{o, o + len(dr.name.Name), a.newName[key]})
		break
	}
	return edits
}

// importEdits adds imports of the named end-state packages and, when drop is
// set, removes the import spec drop.
func (a *analysis) importEdits(f *ast.File, src []byte, need map[string]bool, drop *ast.ImportSpec, fileOff func(token.Pos) int) []edit {
	var paths []string
	have := map[string]bool{}
	for _, s := range f.Imports {
		p, _ := strconv.Unquote(s.Path.Value)
		have[p] = true
	}
	for q := range need {
		p := a.module + "/" + pkgDir[q]
		if !have[p] {
			paths = append(paths, p)
		}
	}
	sort.Strings(paths)
	var edits []edit
	if drop != nil {
		o, e := fileOff(drop.Pos()), fileOff(drop.End())
		for o > 0 && src[o-1] != '\n' {
			o--
		}
		for e < len(src) && src[e] != '\n' {
			e++
		}
		if e < len(src) {
			e++
		}
		edits = append(edits, edit{o, e, ""})
	}
	if len(paths) == 0 {
		return edits
	}
	var lines strings.Builder
	for _, p := range paths {
		fmt.Fprintf(&lines, "\t%q\n", p)
	}
	var imp *ast.GenDecl
	for _, d := range f.Decls {
		if g, ok := d.(*ast.GenDecl); ok && g.Tok == token.IMPORT {
			imp = g
		}
	}
	switch {
	case imp == nil:
		o := fileOff(f.Name.End())
		edits = append(edits, edit{o, o, "\n\nimport (\n" + lines.String() + ")"})
	case !imp.Lparen.IsValid():
		spec := imp.Specs[0].(*ast.ImportSpec)
		o, e := fileOff(spec.Pos()), fileOff(spec.End())
		edits = append(edits, edit{o, e, "(\n\t" + string(src[o:e]) + "\n\n" + lines.String() + ")"})
	default:
		// after the last import of this module, else as a group of its own
		var last *ast.ImportSpec
		for _, s := range imp.Specs {
			is := s.(*ast.ImportSpec)
			p, _ := strconv.Unquote(is.Path.Value)
			if strings.HasPrefix(p, a.module+"/") && is != drop {
				last = is
			}
		}
		if last != nil {
			e := fileOff(last.End())
			for e < len(src) && src[e] != '\n' {
				e++
			}
			e++
			edits = append(edits, edit{e, e, lines.String()})
		} else {
			o := fileOff(imp.Rparen)
			edits = append(edits, edit{o, o, "\n" + lines.String()})
		}
	}
	return edits
}

var testdataRE = regexp.MustCompile(`"testdata/([^/"]+)/`)

// destName is a file's name in its new directory.
func (a *analysis) destName(base, pkg string) string {
	if a.p.helper[pkg] {
		n := strings.TrimPrefix(base, helperPrefix[pkg])
		return strings.TrimSuffix(n, "_test.go") + ".go"
	}
	return base
}

func formatOrFail(name string, b []byte) []byte {
	out, err := format.Source(b)
	if err != nil {
		os.WriteFile("/dev/stderr", b, 0)
		fail("format %s: %v", name, err)
	}
	return out
}

func run(env []string, p *plan) *result {
	all := goList(env)
	var syncPkg *goPkg
	for i := range all {
		if strings.HasSuffix(all[i].ImportPath, "/"+syncRel) {
			syncPkg = &all[i]
		}
	}
	if syncPkg == nil {
		fail("internal/sync not found")
	}
	module := strings.TrimSuffix(syncPkg.ImportPath, "/"+syncRel)
	exports := exportData(env)
	fset := token.NewFileSet()
	imp := importer.ForCompiler(fset, "gc", func(path string) (io.ReadCloser, error) {
		f, ok := exports[path]
		if !ok {
			return nil, fmt.Errorf("no export data for %s", path)
		}
		return os.Open(f)
	})
	parse := func(dir string, names []string) ([]*ast.File, map[string][]byte) {
		var files []*ast.File
		srcs := map[string][]byte{}
		for _, n := range names {
			b, err := os.ReadFile(filepath.Join(dir, n))
			if err != nil {
				fail("%v", err)
			}
			f, err := parser.ParseFile(fset, filepath.Join(dir, n), b, parser.ParseComments)
			if err != nil {
				fail("parse %s: %v", n, err)
			}
			files = append(files, f)
			srcs[n] = b
		}
		return files, srcs
	}
	check := func(path string, files []*ast.File) (*types.Package, *types.Info) {
		info := &types.Info{Uses: map[*ast.Ident]types.Object{}, Defs: map[*ast.Ident]types.Object{}}
		var errs []string
		conf := types.Config{Importer: imp, Error: func(err error) { errs = append(errs, err.Error()) }}
		pkg, _ := conf.Check(path, fset, files, info)
		if len(errs) > 0 {
			fail("type-check %s: %d errors, first: %s", path, len(errs), errs[0])
		}
		return pkg, info
	}

	names := append(append([]string{}, syncPkg.GoFiles...), syncPkg.TestGoFiles...)
	files, srcs := parse(syncPkg.Dir, names)
	pkg, info := check(syncPkg.ImportPath, files)
	a := &analysis{
		p: p, fset: fset, pkg: pkg, info: info, syncPath: syncPkg.ImportPath, module: module,
		files: map[string]*ast.File{}, srcs: srcs, marked: map[string]bool{}, errors: map[string]bool{},
	}
	for _, f := range files {
		a.files[a.fileOf(f.Pos())] = f
	}
	a.buildFieldKeys()

	a.fileHome = func(base string) string {
		h, ok := p.fileHome[base]
		if !ok {
			// a file without declarations of its own (doc.go) or one the simulator did not type-check
			for _, dr := range a.declRanges(a.files[base]) {
				if hh, ok := p.declHome[dr.id]; ok {
					return hh
				}
			}
			return "sync"
		}
		return h
	}

	fileHome := a.fileHome
	// declarations by id, for the private copies
	byID := map[string]declRange{}
	declFile := map[string]string{}
	a.declFile = declFile
	for base, f := range a.files {
		for _, dr := range a.declRanges(f) {
			if _, dup := byID[dr.id]; dup && !strings.HasPrefix(dr.id, "_") {
				fail("declaration %s twice", dr.id)
			}
			byID[dr.id] = dr
			declFile[dr.id] = base
		}
	}
	type copyDecl struct {
		pkg, id string
	}
	var copyList []copyDecl
	for cp, ids := range p.copies {
		for id := range ids {
			if _, ok := byID[id]; !ok {
				fail("testcopy %s %s: no such declaration", cp, id)
			}
			copyList = append(copyList, copyDecl{cp, id})
		}
		// methods follow a copied type
		for id := range byID {
			if i := strings.Index(id, "."); i > 0 && ids[id[:i]] {
				copyList = append(copyList, copyDecl{cp, id})
			}
		}
	}
	sort.Slice(copyList, func(i, j int) bool {
		x, y := copyList[i], copyList[j]
		if x.pkg != y.pkg {
			return x.pkg < y.pkg
		}
		fx, fy := declFile[x.id], declFile[y.id]
		if fx != fy {
			return fx < fy
		}
		return byID[x.id].start < byID[y.id].start
	})

	// 1. marking: every file as its package, every copy as the copying package
	sortedBases := make([]string, 0, len(a.files))
	for b := range a.files {
		sortedBases = append(sortedBases, b)
	}
	sort.Strings(sortedBases)
	for _, base := range sortedBases {
		f := a.files[base]
		a.mark(f, f.Pos(), f.End(), fileHome(base))
	}
	for _, c := range copyList {
		dr := byID[c.id]
		a.mark(a.files[declFile[c.id]], dr.start, dr.end, c.pkg)
	}
	// a capitalised name must not meet one that exists in its package; a field
	// that meets a method of its type takes a suffix that says what it holds
	a.newName = map[string]string{}
	for key := range a.marked {
		i := strings.LastIndex(key, ".")
		if i < 0 {
			up := exported(key)
			a.newName[key] = up
			if o := pkg.Scope().Lookup(up); o != nil && a.home(up, "") == a.home(key, "") {
				a.errors["capitalising "+key+" collides with an existing "+up] = true
			}
			continue
		}
		up := exported(key[i+1:])
		a.newName[key] = up
		tn, _ := pkg.Scope().Lookup(strings.SplitN(key, ".", 2)[0]).(*types.TypeName)
		if tn == nil {
			continue
		}
		o, _, _ := types.LookupFieldOrMethod(tn.Type(), true, pkg, up)
		if o == nil {
			continue
		}
		field, _, _ := types.LookupFieldOrMethod(tn.Type(), true, pkg, key[i+1:])
		v, isVar := field.(*types.Var)
		_, isFunc := o.(*types.Func)
		switch {
		case isVar && isFunc && isSig(v.Type()):
			a.newName[key] = up + "Func"
		case isVar && isFunc && isChan(v.Type()):
			a.newName[key] = up + "Ch"
		default:
			a.errors["capitalising "+key+" collides with "+key[:i+1]+up] = true
			continue
		}
		if o2, _, _ := types.LookupFieldOrMethod(tn.Type(), true, pkg, a.newName[key]); o2 != nil {
			a.errors["capitalising "+key+" collides with "+key[:i+1]+a.newName[key]] = true
		}
	}

	res := &result{files: map[string][]byte{}, source: map[string]string{}, marked: a.marked}
	rel := func(dir, base string) string {
		r, _ := filepath.Rel(*repoFlag, filepath.Join(dir, base))
		return r
	}

	// 2. the files of internal/sync
	for _, base := range sortedBases {
		f := a.files[base]
		loc := fileHome(base)
		edits, need := a.render(f, f.Pos(), f.End(), loc)
		for _, dr := range a.declRanges(f) {
			edits = append(edits, a.docRename(dr, loc)...)
		}
		if loc != "sync" {
			edits = append(edits, edit{a.off(f.Name.Pos()), a.off(f.Name.End()), loc})
		}
		edits = append(edits, a.importEdits(f, srcs[base], need, nil, a.off)...)
		dest := filepath.Join(pkgDir[loc], a.destName(base, loc))
		if _, dup := res.files[dest]; dup {
			fail("two files become %s", dest)
		}
		res.files[dest] = formatOrFail(dest, apply(srcs[base], edits))
		res.source[dest] = rel(syncPkg.Dir, base)
	}

	// 3. the private copies, one file per copying package
	byPkg := map[string][]copyDecl{}
	for _, c := range copyList {
		byPkg[c.pkg] = append(byPkg[c.pkg], c)
	}
	for cp, list := range byPkg {
		var body strings.Builder
		need := map[string]bool{}
		imports := map[string]string{} // path -> local name
		// specs cut out of one group stay one group, under the group's directives
		var open *ast.GenDecl
		closeGroup := func() {
			if open != nil {
				body.WriteString(")\n")
				open = nil
			}
		}
		for _, c := range list {
			dr := byID[c.id]
			f := a.files[declFile[c.id]]
			src := srcs[declFile[c.id]]
			if dr.keyword == "const " {
				if vs, ok := findSpec(f, dr.name).(*ast.ValueSpec); !ok || len(vs.Values) == 0 {
					fail("testcopy %s: a constant without its own value cannot be copied alone", c.id)
				}
			}
			edits, n := a.render(f, dr.start, dr.end, cp)
			for q := range n {
				need[q] = true
			}
			base := a.off(dr.start)
			for i := range edits {
				edits[i].off -= base
				edits[i].end -= base
			}
			text := apply(append([]byte{}, src[base:a.off(dr.end)]...), edits)
			switch {
			case dr.group != nil && dr.group == open:
				body.WriteString("\t")
				body.Write(text)
				body.WriteString("\n")
			case dr.group != nil:
				closeGroup()
				body.WriteString("\n")
				for _, d := range directives(dr.group.Doc) {
					body.WriteString(d + "\n")
				}
				body.WriteString(dr.group.Tok.String() + " (\n\t")
				body.Write(text)
				body.WriteString("\n")
				open = dr.group
			default:
				closeGroup()
				body.WriteString("\n")
				body.Write(text)
				body.WriteString("\n")
			}
			ast.Inspect(f, func(nd ast.Node) bool {
				if nd == nil || nd.End() <= dr.start || nd.Pos() >= dr.end {
					return false
				}
				if id, ok := nd.(*ast.Ident); ok {
					if pn, ok := info.Uses[id].(*types.PkgName); ok {
						imports[pn.Imported().Path()] = pn.Name()
						if pn.Name() != pn.Imported().Name() {
							imports[pn.Imported().Path()] = pn.Name() + " "
						}
					}
				}
				return true
			})
		}
		closeGroup()
		for q := range need {
			imports[module+"/"+pkgDir[q]] = q
		}
		var paths []string
		for path := range imports {
			paths = append(paths, path)
		}
		sort.Strings(paths)
		// the groups goimports keeps: standard library, other modules, this module
		var std, other, local strings.Builder
		for _, path := range paths {
			line := fmt.Sprintf("\t%q\n", path)
			if alias := imports[path]; strings.HasSuffix(alias, " ") {
				line = "\t" + alias + strconv.Quote(path) + "\n"
			}
			switch {
			case path == module || strings.HasPrefix(path, module+"/"):
				local.WriteString(line)
			case strings.Contains(strings.SplitN(path, "/", 2)[0], "."):
				other.WriteString(line)
			default:
				std.WriteString(line)
			}
		}
		var homes []string
		seen := map[string]bool{}
		for _, c := range list {
			h := p.declHome[strings.SplitN(c.id, ".", 2)[0]]
			if !seen[h] {
				seen[h] = true
				homes = append(homes, h)
			}
		}
		sort.Strings(homes)
		var groups []string
		for _, g := range []string{std.String(), other.String(), local.String()} {
			if g != "" {
				groups = append(groups, g)
			}
		}
		header := wrapComment("Private copies of test helpers whose home ("+strings.Join(homes, ", ")+
			") these tests cannot import, because it imports this package or is another package's tests.", 76)
		src := fmt.Sprintf("package %s\n\n%s\nimport (\n%s)\n%s", cp, header, strings.Join(groups, "\n"), body.String())
		dest := filepath.Join(pkgDir[cp], "helper_copies_test.go")
		if _, dup := res.files[dest]; dup {
			fail("%s exists", dest)
		}
		res.files[dest] = formatOrFail(dest, []byte(src))
	}

	// 4. testdata named by one package's tests
	tdHome := map[string]string{}
	for _, base := range sortedBases {
		if !strings.HasSuffix(base, "_test.go") {
			continue
		}
		for _, m := range testdataRE.FindAllSubmatch(srcs[base], -1) {
			d, loc := string(m[1]), fileHome(base)
			if old, ok := tdHome[d]; ok && old != loc {
				a.errors[fmt.Sprintf("testdata/%s is named by %s and %s", d, old, loc)] = true
			}
			tdHome[d] = loc
		}
	}
	for d, loc := range tdHome {
		if loc == "sync" {
			continue
		}
		res.source[filepath.Join(pkgDir[loc], "testdata", d)] = rel(syncPkg.Dir, filepath.Join("testdata", d))
	}

	// 5. the other packages that import internal/sync
	for _, gp := range all {
		if gp.ImportPath == syncPkg.ImportPath || !(contains(gp.Imports, a.syncPath) || contains(gp.TestImports, a.syncPath)) {
			continue
		}
		if len(gp.XTestGoFiles) > 0 && contains(gp.XTestImports, a.syncPath) {
			fail("%s: external test packages are not handled", gp.ImportPath)
		}
		cfiles, csrcs := parse(gp.Dir, append(append([]string{}, gp.GoFiles...), gp.TestGoFiles...))
		_, cinfo := check(gp.ImportPath, cfiles)
		for _, f := range cfiles {
			base := filepath.Base(fset.Position(f.Pos()).Filename)
			var edits []edit
			need := map[string]bool{}
			keep := false
			var syncSpec *ast.ImportSpec
			for _, s := range f.Imports {
				if pth, _ := strconv.Unquote(s.Path.Value); pth == a.syncPath {
					syncSpec = s
				}
			}
			if syncSpec == nil {
				continue
			}
			ast.Inspect(f, func(n ast.Node) bool {
				sel, ok := n.(*ast.SelectorExpr)
				if !ok {
					return true
				}
				x, ok := sel.X.(*ast.Ident)
				if !ok {
					return true
				}
				pn, ok := cinfo.Uses[x].(*types.PkgName)
				if !ok || pn.Imported().Path() != a.syncPath {
					return true
				}
				h := a.home(sel.Sel.Name, "")
				if h == "sync" {
					keep = true
					return true
				}
				if pkgDir[h] != "" && strings.HasPrefix(pkgDir[h], syncRel+"/internal/") {
					a.errors[fmt.Sprintf("%s: %s.%s moves to private package %s", fset.Position(sel.Pos()), x.Name, sel.Sel.Name, h)] = true
				}
				need[h] = true
				edits = append(edits, edit{fset.Position(x.Pos()).Offset, fset.Position(x.End()).Offset, h})
				return true
			})
			if len(edits) == 0 {
				continue
			}
			var drop *ast.ImportSpec
			if !keep {
				drop = syncSpec
			}
			edits = append(edits, a.importEdits(f, csrcs[base], need, drop, func(pos token.Pos) int { return fset.Position(pos).Offset })...)
			dest := rel(gp.Dir, base)
			res.files[dest] = formatOrFail(dest, apply(csrcs[base], edits))
			res.source[dest] = dest
		}
	}
	for e := range a.errors {
		res.errors = append(res.errors, e)
	}
	sort.Strings(res.errors)
	return res
}

// wrapComment renders text as // lines of at most width columns.
func wrapComment(text string, width int) string {
	var b, line strings.Builder
	for _, w := range strings.Fields(text) {
		if line.Len() > 0 && line.Len()+1+len(w) > width-3 {
			b.WriteString("// " + line.String() + "\n")
			line.Reset()
		}
		if line.Len() > 0 {
			line.WriteString(" ")
		}
		line.WriteString(w)
	}
	if line.Len() > 0 {
		b.WriteString("// " + line.String() + "\n")
	}
	return b.String()
}

func findSpec(f *ast.File, name *ast.Ident) ast.Spec {
	for _, d := range f.Decls {
		if g, ok := d.(*ast.GenDecl); ok {
			for _, s := range g.Specs {
				if vs, ok := s.(*ast.ValueSpec); ok {
					for _, n := range vs.Names {
						if n == name {
							return vs
						}
					}
				}
			}
		}
	}
	return nil
}

func contains(xs []string, x string) bool {
	for _, y := range xs {
		if y == x {
			return true
		}
	}
	return false
}

func main() {
	flag.Parse()
	abs, err := filepath.Abs(*repoFlag)
	if err != nil {
		fail("%v", err)
	}
	*repoFlag = abs
	p := loadPlan()
	linux := run([]string{"GOOS=linux", "GOARCH=amd64"}, p)
	darwin := run([]string{"GOOS=darwin", "GOARCH=arm64", "CGO_ENABLED=0"}, p)
	errs := append(append([]string{}, linux.errors...), darwin.errors...)
	// one rewrite per file: a file both file sets type-check must agree
	out := linux
	for dest, b := range darwin.files {
		if lb, ok := out.files[dest]; ok {
			if !bytes.Equal(lb, b) {
				errs = append(errs, "the Linux and macOS rewrites of "+dest+" differ")
			}
			continue
		}
		out.files[dest] = b
		out.source[dest] = darwin.source[dest]
	}
	for dest, src := range darwin.source {
		if _, ok := out.source[dest]; !ok {
			out.source[dest] = src
		}
	}
	for k := range darwin.marked {
		if !linux.marked[k] {
			errs = append(errs, "capitalised only for macOS: "+k)
		}
	}
	// files neither set type-checks keep their content apart from the package clause
	syncDir := filepath.Join(*repoFlag, syncRel)
	entries, err := os.ReadDir(syncDir)
	if err != nil {
		fail("%v", err)
	}
	moved := map[string]bool{}
	for _, src := range out.source {
		moved[src] = true
	}
	for _, e := range entries {
		base := e.Name()
		if e.IsDir() || !strings.HasSuffix(base, ".go") || moved[filepath.Join(syncRel, base)] {
			continue
		}
		b, err := os.ReadFile(filepath.Join(syncDir, base))
		if err != nil {
			fail("%v", err)
		}
		fset := token.NewFileSet()
		f, err := parser.ParseFile(fset, base, b, parser.ParseComments)
		if err != nil {
			fail("parse %s: %v", base, err)
		}
		loc := ""
		for _, d := range f.Decls {
			if g, ok := d.(*ast.GenDecl); ok {
				for _, s := range g.Specs {
					if vs, ok := s.(*ast.ValueSpec); ok {
						for _, n := range vs.Names {
							if h := p.declHome[n.Name]; h != "" {
								loc = h
							}
						}
					}
				}
			}
		}
		if loc == "" {
			errs = append(errs, "no package for the unchecked file "+base)
			continue
		}
		// such a file may name nothing of the sync tree: only constants of basic literals
		ast.Inspect(f, func(n ast.Node) bool {
			if id, ok := n.(*ast.Ident); ok && id != f.Name && id.Obj == nil && id.Name != "true" && id.Name != "false" && id.Name != "bool" {
				errs = append(errs, fmt.Sprintf("unchecked file %s names %s", base, id.Name))
			}
			return true
		})
		pos := fset.Position(f.Name.Pos()).Offset
		nb := apply(b, []edit{{pos, pos + len(f.Name.Name), loc}})
		dest := filepath.Join(pkgDir[loc], strings.TrimPrefix(strings.TrimSuffix(base, "_test.go")+".go", helperPrefix[loc]))
		if !p.helper[loc] {
			dest = filepath.Join(pkgDir[loc], base)
		}
		out.files[dest] = formatOrFail(dest, nb)
		out.source[dest] = filepath.Join(syncRel, base)
	}
	if len(errs) > 0 {
		for _, e := range errs {
			fmt.Fprintln(os.Stderr, e)
		}
		fail("%d problems; nothing written", len(errs))
	}

	dests := make([]string, 0, len(out.source))
	for d := range out.source {
		dests = append(dests, d)
	}
	for d := range out.files {
		if _, ok := out.source[d]; !ok {
			dests = append(dests, d)
		}
	}
	sort.Strings(dests)
	perPkg := map[string]int{}
	for _, d := range dests {
		perPkg[filepath.Dir(d)]++
	}
	var markedKeys []string
	for k := range out.marked {
		markedKeys = append(markedKeys, k)
	}
	sort.Strings(markedKeys)
	fmt.Printf("%d destinations; %d helper identifiers capitalised\n", len(dests), len(markedKeys))
	var dirs []string
	for d := range perPkg {
		dirs = append(dirs, d)
	}
	sort.Strings(dirs)
	for _, d := range dirs {
		fmt.Printf("  %-40s %d\n", d, perPkg[d])
	}
	if *dryFlag {
		for _, k := range markedKeys {
			fmt.Println("  capitalise", k)
		}
		return
	}
	// write: directories and testdata first, then every file; then remove the sources that moved
	for _, d := range dests {
		src, isMove := out.source[d]
		abs := filepath.Join(*repoFlag, d)
		if err := os.MkdirAll(filepath.Dir(abs), 0o755); err != nil {
			fail("%v", err)
		}
		if b, ok := out.files[d]; ok {
			if err := os.WriteFile(abs, b, 0o644); err != nil {
				fail("%v", err)
			}
			if isMove && src != d {
				if err := os.Remove(filepath.Join(*repoFlag, src)); err != nil {
					fail("%v", err)
				}
			}
			continue
		}
		if err := os.Rename(filepath.Join(*repoFlag, src), abs); err != nil {
			fail("%v", err)
		}
	}
}
