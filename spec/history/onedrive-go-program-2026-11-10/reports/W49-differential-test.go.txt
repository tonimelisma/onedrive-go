package sync

// Differential harness (scratch only, never committed). It builds a small
// synced tree, applies every valid sequence of up to four steps (and a seeded
// sample of five-step sequences), on this computer or on OneDrive, syncs at
// most three times, and writes one line per sequence and side to the file
// named by DW_OUT. Trees are written by digest to DW_OUT + ".trees".
//
// Tree: folders A (holding a.txt and subfolder s, which holds t.txt) and B
// (holding b.txt).
//
// Steps, each folder step at most once per sequence:
//   renA  rename folder A (the synced item, wherever it is) to A2
//   renS  rename folder s to s2
//   mvBA  move folder B into the folder now at path A
//   mvSB  move folder s into the folder now at path B
//   mk    make a new folder at the path the latest folder step left
//   mv<f><o|n>  move file f (a, t, b) into the newest folder made by mk,
//         under its own name (o) or a new one (n)
//   bk<f> move file f back to the path its own last move left

import (
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"log/slog"
	"math/rand/v2"
	"os"
	"path"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
	stdsync "sync"
	"testing"

	"github.com/tonimelisma/onedrive-go/internal/driveid"
)

type dwState struct {
	paths    map[string]string // entity -> current path
	folder   map[string]bool
	prev     map[string]string // file -> path before its own last move
	vacated  string
	newest   string
	used     map[string]bool
	madeSeen int
}

func dwInitial() *dwState {
	s := &dwState{
		paths: map[string]string{
			"A": "A", "s": "A/s", "B": "B", "a": "A/a.txt", "t": "A/s/t.txt", "b": "B/b.txt",
		},
		folder: map[string]bool{"A": true, "s": true, "B": true},
		prev:   map[string]string{},
		used:   map[string]bool{},
	}

	return s
}

func (s *dwState) clone() *dwState {
	c := &dwState{
		paths: map[string]string{}, folder: map[string]bool{}, prev: map[string]string{},
		used: map[string]bool{}, vacated: s.vacated, newest: s.newest, madeSeen: s.madeSeen,
	}
	for k, v := range s.paths {
		c.paths[k] = v
	}
	for k, v := range s.folder {
		c.folder[k] = v
	}
	for k, v := range s.prev {
		c.prev[k] = v
	}
	for k, v := range s.used {
		c.used[k] = v
	}

	return c
}

func (s *dwState) at(p string) (string, bool) {
	for id, cur := range s.paths {
		if cur == p {
			return id, true
		}
	}

	return "", false
}

func (s *dwState) folderAt(p string) bool {
	if p == "." || p == "" {
		return true
	}
	id, ok := s.at(p)

	return ok && s.folder[id]
}

func (s *dwState) move(id, to string) {
	from := s.paths[id]
	for other, cur := range s.paths {
		switch {
		case cur == from:
			s.paths[other] = to
		case strings.HasPrefix(cur, from+"/"):
			s.paths[other] = to + strings.TrimPrefix(cur, from)
		}
	}
}

// dwStep is one valid step: its code and the change it makes.
type dwStep struct {
	code     string
	from, to string // a move from -> to, or a mkdir at to when from is ""
}

func dwJoin(dir, name string) string {
	if dir == "." || dir == "" {
		return name
	}

	return dir + "/" + name
}

func (s *dwState) folderStep(code, id, dest string) (dwStep, bool) {
	if s.used[code] {
		return dwStep{}, false
	}
	from := s.paths[id]
	if dest == from || strings.HasPrefix(dest, from+"/") {
		return dwStep{}, false
	}
	if _, taken := s.at(dest); taken || !s.folderAt(path.Dir(dest)) {
		return dwStep{}, false
	}

	return dwStep{code: code, from: from, to: dest}, true
}

func (s *dwState) steps() []dwStep {
	var out []dwStep
	add := func(st dwStep, ok bool) {
		if ok {
			out = append(out, st)
		}
	}
	add(s.folderStep("renA", "A", dwJoin(path.Dir(s.paths["A"]), "A2")))
	add(s.folderStep("renS", "s", dwJoin(path.Dir(s.paths["s"]), "s2")))
	if s.folderAt("A") && path.Dir(s.paths["B"]) != "A" {
		add(s.folderStep("mvBA", "B", "A/"+path.Base(s.paths["B"])))
	}
	if s.folderAt("B") && path.Dir(s.paths["s"]) != "B" {
		add(s.folderStep("mvSB", "s", "B/"+path.Base(s.paths["s"])))
	}
	if s.vacated != "" && s.madeSeen < 2 {
		if _, taken := s.at(s.vacated); !taken && s.folderAt(path.Dir(s.vacated)) {
			out = append(out, dwStep{code: "mk", to: s.vacated})
		}
	}
	for _, f := range []string{"a", "t", "b"} {
		cur := s.paths[f]
		if s.newest != "" {
			dir := s.paths[s.newest]
			for _, own := range []bool{true, false} {
				code := "mv" + f + "n"
				name := "n" + path.Base(cur)
				if own {
					code, name = "mv"+f+"o", path.Base(cur)
				}
				dest := dir + "/" + name
				if _, taken := s.at(dest); !s.used[code] && path.Dir(cur) != dir && !taken {
					out = append(out, dwStep{code: code, from: cur, to: dest})
				}
			}
		}
		if back := s.prev[f]; back != "" && back != cur && !s.used["bk"+f] {
			if _, taken := s.at(back); !taken && s.folderAt(path.Dir(back)) {
				out = append(out, dwStep{code: "bk" + f, from: cur, to: back})
			}
		}
	}

	return out
}

func (s *dwState) apply(st dwStep) {
	s.used[st.code] = true
	switch {
	case st.code == "mk":
		s.madeSeen++
		id := "N" + strconv.Itoa(s.madeSeen)
		s.paths[id] = st.to
		s.folder[id] = true
		s.newest = id
		s.vacated = ""
	case strings.HasPrefix(st.code, "mv") && len(st.code) == 4 && !strings.HasPrefix(st.code, "mvBA") && !strings.HasPrefix(st.code, "mvSB"):
		f := st.code[2:3]
		s.prev[f] = s.paths[f]
		s.move(f, st.to)
	case strings.HasPrefix(st.code, "bk"):
		f := st.code[2:3]
		s.prev[f] = s.paths[f]
		s.move(f, st.to)
	default:
		id, _ := s.at(st.from)
		s.move(id, st.to)
		s.vacated = st.from
	}
}

// dwSequences lists every valid sequence of 1 to maxLen steps, in order.
func dwSequences(s *dwState, prefix []dwStep, maxLen int, out *[][]dwStep) {
	if len(prefix) > 0 {
		*out = append(*out, append([]dwStep(nil), prefix...))
	}
	if len(prefix) == maxLen {
		return
	}
	for _, st := range s.steps() {
		next := s.clone()
		next.apply(st)
		dwSequences(next, append(prefix, st), maxLen, out)
	}
}

func dwCodes(seq []dwStep) string {
	codes := make([]string, len(seq))
	for i := range seq {
		codes[i] = seq[i].code
	}

	return strings.Join(codes, ",")
}

func dwTreeText(tree map[string]string) string {
	keys := make([]string, 0, len(tree))
	for k := range tree {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	var b strings.Builder
	for _, k := range keys {
		fmt.Fprintf(&b, "%s=%s;", k, strings.TrimSpace(tree[k]))
	}

	return b.String()
}

func dwDigest(text string) string {
	sum := sha256.Sum256([]byte(text))

	return hex.EncodeToString(sum[:6])
}

type dwResults struct {
	mu    stdsync.Mutex
	lines []string
	trees map[string]string
}

func (r *dwResults) add(line string, trees map[string]string) {
	r.mu.Lock()
	defer r.mu.Unlock()
	r.lines = append(r.lines, line)
	for k, v := range trees {
		r.trees[k] = v
	}
}

func dwRun(t *testing.T, seq []dwStep, onOneDrive bool) (string, map[string]string) {
	t.Helper()

	remote := newMoveScenarioRemote(driveid.New(engineTestDriveID))
	remote.addFolder("item-A", "root", "A")
	remote.addFile(t, "item-a", "item-A", "a.txt", "a\n")
	remote.addFolder("item-s", "item-A", "s")
	remote.addFile(t, "item-t", "item-s", "t.txt", "t\n")
	remote.addFolder("item-B", "root", "B")
	remote.addFile(t, "item-b", "item-B", "b.txt", "b\n")
	logs := &syncLockedBuffer{}
	logger := slog.New(slog.NewTextHandler(logs, &slog.HandlerOptions{Level: slog.LevelError}))
	eng, syncRoot := newTestEngineWithLogger(t, remote.client(t), logger)
	runScenarioPass(t, eng)
	local := func(rel string) string { return filepath.Join(syncRoot, filepath.FromSlash(rel)) }

	stepErr := ""
	for _, st := range seq {
		switch {
		case onOneDrive && st.from == "":
			remote.mkdirAt(t, st.to)
		case onOneDrive:
			remote.moveAt(t, st.from, st.to)
		case st.from == "":
			if err := os.Mkdir(local(st.to), 0o750); err != nil {
				stepErr = err.Error()
			}
		default:
			if err := os.Rename(local(st.from), local(st.to)); err != nil {
				stepErr = err.Error()
			}
		}
	}

	runs, superseded := 0, 0
	done := false
	for run := 1; run <= 3; run++ {
		report := runScenarioPass(t, eng)
		superseded += report.Superseded
		runs = run
		if !report.LeftWork().Any() {
			done = true
			break
		}
	}
	localText := dwTreeText(harnessLocalTree(t, syncRoot))
	remoteText := dwTreeText(remote.tree())
	side := "here"
	if onOneDrive {
		side = "onedrive"
	}
	localDigest, remoteDigest := dwDigest(localText), dwDigest(remoteText)
	line := fmt.Sprintf("%s\t%s\tdone=%v\truns=%d\tsuperseded=%d\tcycles=%d\tlocal=%s\tremote=%s\tequal=%v\terr=%s",
		side, dwCodes(seq), done, runs, superseded, strings.Count(logs.String(), "dependency cycle"),
		localDigest, remoteDigest, localText == remoteText, stepErr)

	return line, map[string]string{localDigest: localText, remoteDigest: remoteText}
}

func TestZZDiff(t *testing.T) {
	out := os.Getenv("DW_OUT")
	if out == "" {
		t.Skip("DW_OUT not set")
	}
	t.Parallel()

	var seqs [][]dwStep
	dwSequences(dwInitial(), nil, 4, &seqs)
	upToFour := len(seqs)
	var five [][]dwStep
	dwSequences(dwInitial(), nil, 5, &five)
	var fiveOnly [][]dwStep
	for _, seq := range five {
		if len(seq) == 5 {
			fiveOnly = append(fiveOnly, seq)
		}
	}
	rng := rand.New(rand.NewPCG(4949, 7)) //nolint:gosec // deterministic sample
	rng.Shuffle(len(fiveOnly), func(i, j int) { fiveOnly[i], fiveOnly[j] = fiveOnly[j], fiveOnly[i] })
	// Five-step sequences that make two folders again come first: they
	// hold the shape of a folder moved into a path another folder left.
	twoMade := 0
	sort.SliceStable(fiveOnly, func(i, j int) bool {
		return strings.Count(dwCodes(fiveOnly[i]), "mk") > strings.Count(dwCodes(fiveOnly[j]), "mk")
	})
	for _, seq := range fiveOnly {
		if strings.Count(dwCodes(seq), "mk") == 2 {
			twoMade++
		}
	}
	fiveLimit, _ := strconv.Atoi(os.Getenv("DW_FIVE"))
	fiveLimit = min(fiveLimit, len(fiveOnly))
	seqs = append(seqs, fiveOnly[:fiveLimit]...)
	if maxSeqs, err := strconv.Atoi(os.Getenv("DW_MAX")); err == nil && maxSeqs < len(seqs) {
		seqs = seqs[:maxSeqs]
	}

	results := &dwResults{trees: map[string]string{}}
	t.Cleanup(func() {
		sort.Strings(results.lines)
		header := fmt.Sprintf("# %d sequences up to 4 steps, %d of %d five-step ones (%d with two mk first, then seed 4949), %d run\n",
			upToFour, fiveLimit, len(fiveOnly), twoMade, len(seqs))
		if err := os.WriteFile(out, []byte(header+strings.Join(results.lines, "\n")+"\n"), 0o600); err != nil {
			panic(err)
		}
		var trees []string
		for k, v := range results.trees {
			trees = append(trees, k+"\t"+v)
		}
		sort.Strings(trees)
		if err := os.WriteFile(out+".trees", []byte(strings.Join(trees, "\n")+"\n"), 0o600); err != nil {
			panic(err)
		}
	})
	for i, seq := range seqs {
		for _, onOneDrive := range []bool{false, true} {
			t.Run(fmt.Sprintf("%d-%v", i, onOneDrive), func(t *testing.T) {
				t.Parallel()
				line, trees := dwRun(t, seq, onOneDrive)
				results.add(line, trees)
			})
		}
	}
}
