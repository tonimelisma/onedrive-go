func zzReviewRun(t *testing.T, name string, remote *moveScenarioRemote, steps func(local func(string) string)) {
	logs := &syncLockedBuffer{}
	logger := slog.New(slog.NewTextHandler(logs, &slog.HandlerOptions{Level: slog.LevelError}))
	eng, syncRoot := newTestEngineWithLogger(t, remote.client(t), logger)
	runScenarioPass(t, eng)

	local := func(rel string) string { return filepath.Join(syncRoot, filepath.FromSlash(rel)) }
	steps(local)

	runs, superseded := 0, 0
	var left LeftWork
	for run := 1; run <= 3; run++ {
		report := runScenarioPass(t, eng)
		superseded += report.Superseded
		runs = run
		left = report.LeftWork()
		if !left.Any() {
			break
		}
	}
	var b strings.Builder
	fmt.Fprintf(&b, "case %s\nruns=%d superseded=%d leftAny=%v left=%+v\n", name, runs, superseded, left.Any(), left)
	fmt.Fprintf(&b, "cycleLines=%d\n", strings.Count(logs.String(), "dependency cycle"))
	for _, line := range strings.Split(logs.String(), "\n") {
		if strings.Contains(line, "dependency cycle") {
			fmt.Fprintf(&b, "  log: %.300s\n", line)
			break
		}
	}
	fmt.Fprintf(&b, "remote=%v\nlocal=%v\nmoves=%v deletes=%v creates=%v\n", remote.tree(), harnessLocalTree(t, syncRoot),
		remote.moves, remote.deletes, remote.folderCreates)
	out := os.Getenv("ZZ_REVIEW_OUT")
	require.NotEmpty(t, out)
	require.NoError(t, os.WriteFile(filepath.Join(out, name+".txt"), []byte(b.String()), 0o600))
}

func zzReviewRemote(t *testing.T) *moveScenarioRemote {
	remote := newMoveScenarioRemote(driveid.New(engineTestDriveID))
	remote.addFolder("item-docs", "root", "docs")
	remote.addFolder("item-sub", "item-docs", "sub")
	remote.addFile(t, "item-inner", "item-sub", "inner.txt", "inner\n")
	remote.addFile(t, "item-kept", "item-docs", "kept.txt", "kept\n")
	remote.addFolder("item-photos", "root", "photos")
	remote.addFile(t, "item-photo", "item-photos", "photo.jpg", "photo\n")
	return remote
}

// On this computer: sub renamed inside docs, docs moved, new docs made, and a
// file of the renamed sub moved into the new docs.
func TestZZReview_NestedRename(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "nested-rename", remote, func(local func(string) string) {
		require.NoError(t, os.Rename(local("docs/sub"), local("docs/sub2")))
		require.NoError(t, os.Rename(local("docs"), local("photos/photos")))
		require.NoError(t, os.Mkdir(local("docs"), 0o750))
		require.NoError(t, os.Rename(local("photos/photos/sub2/inner.txt"), local("docs/inner.txt")))
	})
}

// The same without the rename of sub (control).
func TestZZReview_NoRename(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "no-rename", remote, func(local func(string) string) {
		require.NoError(t, os.Rename(local("docs"), local("photos/photos")))
		require.NoError(t, os.Mkdir(local("docs"), 0o750))
		require.NoError(t, os.Rename(local("photos/photos/sub/inner.txt"), local("docs/inner.txt")))
	})
}

// On this computer: docs moved to archive, a new docs made, inner.txt moved
// from archive/sub into it and archive/sub removed; on OneDrive a new file
// is added to docs/sub.
func TestZZReview_OneSidedRecreated(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "one-sided-recreated", remote, func(local func(string) string) {
		require.NoError(t, os.Rename(local("docs"), local("archive")))
		require.NoError(t, os.Mkdir(local("docs"), 0o750))
		require.NoError(t, os.Rename(local("archive/sub/inner.txt"), local("docs/inner.txt")))
		require.NoError(t, os.Remove(local("archive/sub")))
		remote.addFile(t, "item-new", "item-sub", "new.txt", "new\n")
	})
}

// The same without the change on OneDrive (control: the delete flavor).
func TestZZReview_OneSidedDeleted(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "one-sided-deleted", remote, func(local func(string) string) {
		require.NoError(t, os.Rename(local("docs"), local("archive")))
		require.NoError(t, os.Mkdir(local("docs"), 0o750))
		require.NoError(t, os.Rename(local("archive/sub/inner.txt"), local("docs/inner.txt")))
		require.NoError(t, os.Remove(local("archive/sub")))
	})
}
