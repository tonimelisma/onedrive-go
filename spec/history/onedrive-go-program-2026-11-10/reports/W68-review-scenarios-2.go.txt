package sync

import (
	"os"
	"testing"

	"github.com/stretchr/testify/require"
)

// On OneDrive: sub renamed inside docs, docs moved, new docs made, and a file
// of the renamed sub moved into the new docs.
func TestZZReview2_NestedRenameOnOneDrive(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "nested-rename-onedrive", remote, func(local func(string) string) {
		remote.moveAt(t, "docs/sub", "docs/sub2")
		remote.moveAt(t, "docs", "photos/photos")
		remote.mkdirAt(t, "docs")
		remote.moveAt(t, "photos/photos/sub2/inner.txt", "docs/inner.txt")
	})
}

// On OneDrive docs moved to archive and archive/kept.txt moved onto docs;
// here kept.txt moved into a new folder docs/new.
func TestZZReview2_OntoVariant(t *testing.T) {
	remote := zzReviewRemote(t)
	zzReviewRun(t, "onto-variant", remote, func(local func(string) string) {
		remote.moveAt(t, "docs", "archive")
		remote.moveAt(t, "archive/kept.txt", "docs")
		require.NoError(t, os.Mkdir(local("docs/new"), 0o750))
		require.NoError(t, os.Rename(local("docs/kept.txt"), local("docs/new/kept.txt")))
	})
}
