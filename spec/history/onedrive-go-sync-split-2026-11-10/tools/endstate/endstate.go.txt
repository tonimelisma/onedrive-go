// endstate simulates a proposed package split of internal/sync at the
// declaration level and reports whether it is legal Go: no import cycle, no
// reference outside the allowed imports, every method in its receiver's
// package, and the full list of identifiers that have to be exported.
//
//	go -C <repo> run /path/endstate.go -repo <repo> -conf conf.txt -out DIR
//
// Stdlib only.

func set(xs ...string) map[string]bool {
	m := map[string]bool{}
	for _, x := range xs {
		m[x] = true
	}
	return m
}

func hasAnyPrefix(s string, ps ...string) bool {
	for _, p := range ps {
		if strings.HasPrefix(s, p) {
			return true
		}
	}
	return false
}

func family(f string) string {
	switch {
	case domainExact[f] || strings.HasPrefix(f, "domain_"):
		return "domain"
	case hasAnyPrefix(f, "observer", "scanner", "socketio", "item_converter") || f == "local_hash_reuse.go":
		return "observe"
	case strings.HasPrefix(f, "planner"):
		return "plan"
	case hasAnyPrefix(f, "executor", "worker", "dep_graph", "active_scopes") || f == "action_freshness.go" || f == "scope.go":
		return "exec"
	case strings.HasPrefix(f, "store") || storeExact[f]:
		return "store"
	case f == "protected_roots.go" || strings.HasPrefix(f, "shortcut_"):
		return "shortcut"
	case strings.HasPrefix(f, "engine") || f == "api_types.go" || f == "observation_findings.go" || f == "permissions.go" || strings.HasPrefix(f, "permission_"):
		return "engine"
	}
	if docsObserve[f] {
		return "observe"
	}
	if docsEngine[f] {
		return "engine"
	}
	return "other"
}

// ---------- config ----------

type config struct {
	famPkg      map[string]string
	filePkg     map[string]string
	declPkg     map[string]string
	toFunc      map[string]string
	allow       map[string]map[string]bool
	order       []string
	testFilePkg map[string]string
	testExport  map[string]bool            // symbols exported only because a test in another package needs them
	viaIface    map[string]map[string]bool // consumer package -> concrete type it sees only through its own interface
	helperPkg   map[string]bool            // ordinary packages that hold shared test support code
	testDeclPkg map[string]string          // test declaration -> package (a helper package, or another package's tests)
	testCopy    map[string]map[string]bool // package -> helper declaration its own tests keep a private copy of
}

func loadConfig(path string) *config {
	c := &config{famPkg: map[string]string{}, filePkg: map[string]string{}, declPkg: map[string]string{}, toFunc: map[string]string{},
		allow: map[string]map[string]bool{}, testFilePkg: map[string]string{}, testExport: map[string]bool{},
		viaIface: map[string]map[string]bool{}, helperPkg: map[string]bool{}, testDeclPkg: map[string]string{},
		testCopy: map[string]map[string]bool{}}
	data, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	for ln, line := range strings.Split(string(data), "\n") {
		if i := strings.Index(line, "#"); i >= 0 {
			line = line[:i]
		}
		f := strings.Fields(line)
		if len(f) == 0 {
			continue
		}
		bad := func() {
			fmt.Fprintf(os.Stderr, "conf line %d: %q\n", ln+1, line)
			os.Exit(1)
		}
		switch f[0] {
		case "order":
			c.order = f[1:]
		case "pkg":
			if len(f) != 3 {
				bad()
			}
			c.famPkg[f[1]] = f[2]
		case "file":
			if len(f) != 3 {
				bad()
			}
			c.filePkg[f[1]] = f[2]
		case "decl":
			if len(f) < 3 {
				bad()
			}
			for _, d := range f[2:] {
				c.declPkg[d] = f[1]
			}
		case "tofunc":
			if len(f) < 3 {
				bad()
			}
			for _, d := range f[2:] {
				c.toFunc[d] = f[1]
			}
		case "allow":
			if len(f) < 2 {
				bad()
			}
			if c.allow[f[1]] == nil {
				c.allow[f[1]] = map[string]bool{}
			}
			for _, d := range f[2:] {
				c.allow[f[1]][d] = true
			}
		case "testfile":
			if len(f) < 3 {
				bad()
			}
			for _, d := range f[2:] {
				c.testFilePkg[d] = f[1]
			}
		case "testexport":
			for _, d := range f[1:] {
				c.testExport[d] = true
			}
		case "viaiface":
			if len(f) < 3 {
				bad()
			}
			if c.viaIface[f[1]] == nil {
				c.viaIface[f[1]] = map[string]bool{}
			}
			for _, d := range f[2:] {
				c.viaIface[f[1]][d] = true
			}
		case "helper":
			for _, d := range f[1:] {
				c.helperPkg[d] = true
			}
		case "testdecl":
			if len(f) < 3 {
				bad()
			}
			for _, d := range f[2:] {
				c.testDeclPkg[d] = f[1]
			}
		case "testcopy":
			if len(f) < 3 {
				bad()
			}
			if c.testCopy[f[1]] == nil {
				c.testCopy[f[1]] = map[string]bool{}
			}
			for _, d := range f[2:] {
				c.testCopy[f[1]][d] = true
			}
		default:
			bad()
		}
	}
	return c
}

// ---------- go list helpers ----------

type goPkg struct {
	Dir, ImportPath, Name                              string
	GoFiles, TestGoFiles, XTestGoFiles, IgnoredGoFiles []string
	Imports, TestImports, XTestImports                 []string
}

func goCmd(args ...string) []byte {
	cmd := exec.Command("nice", append([]string{"go"}, args...)...)
	cmd.Dir = *repoFlag
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	out, err := cmd.Output()
	if err != nil {
		fmt.Fprintf(os.Stderr, "go %s: %v\n%s\n", strings.Join(args, " "), err, stderr.String())
		os.Exit(1)
	}
	return out
}

func goListJSON(args ...string) []goPkg {
	out := goCmd(append([]string{"list", "-json", "-tags", tags}, args...)...)
	dec := json.NewDecoder(bytes.NewReader(out))
	var pkgs []goPkg
	for {
		var p goPkg
		if err := dec.Decode(&p); err == io.EOF {
			break
		} else if err != nil {
			panic(err)
		}
		pkgs = append(pkgs, p)
	}
	return pkgs
}

// ---------- resolver ----------

type resolver struct {
	pkg        *types.Package
	fieldOwner map[*types.Var]*types.TypeName
	imethOwner map[*types.Func]*types.TypeName
}

func newResolver(pkg *types.Package) *resolver {
	r := &resolver{pkg: pkg, fieldOwner: map[*types.Var]*types.TypeName{}, imethOwner: map[*types.Func]*types.TypeName{}}
	scope := pkg.Scope()
	for _, n := range scope.Names() {
		tn, ok := scope.Lookup(n).(*types.TypeName)
		if !ok || tn.IsAlias() {
			continue
		}
		named, ok := tn.Type().(*types.Named)
		if !ok {
			continue
		}
		r.walk(named.Underlying(), tn, 0)
	}
	return r
}

func (r *resolver) walk(t types.Type, owner *types.TypeName, depth int) {
	if depth > 8 {
		return
	}
	switch u := t.(type) {
	case *types.Struct:
		for i := 0; i < u.NumFields(); i++ {
			f := u.Field(i)
			r.fieldOwner[f] = owner
			r.walk(f.Type(), owner, depth+1)
		}
	case *types.Pointer:
		r.walk(u.Elem(), owner, depth+1)
	case *types.Slice:
		r.walk(u.Elem(), owner, depth+1)
	case *types.Array:
		r.walk(u.Elem(), owner, depth+1)
	case *types.Chan:
		r.walk(u.Elem(), owner, depth+1)
	case *types.Map:
		r.walk(u.Key(), owner, depth+1)
		r.walk(u.Elem(), owner, depth+1)
	case *types.Interface:
		for i := 0; i < u.NumExplicitMethods(); i++ {
			r.imethOwner[u.ExplicitMethod(i)] = owner
		}
	case *types.Signature:
		for _, tup := range []*types.Tuple{u.Params(), u.Results()} {
			for i := 0; i < tup.Len(); i++ {
				r.walk(tup.At(i).Type(), owner, depth+1)
			}
		}
	}
}

type symRef struct {
	name, kind, owner, member string
}

func recvNamed(t types.Type) *types.TypeName {
	for {
		switch u := t.(type) {
		case *types.Pointer:
			t = u.Elem()
			continue
		case *types.Named:
			return u.Origin().Obj()
		case *types.Alias:
			t = types.Unalias(u)
			continue
		}
		return nil
	}
}

func (r *resolver) resolve(obj types.Object) (symRef, bool) {
	if obj == nil || obj.Pkg() != r.pkg {
		return symRef{}, false
	}
	scope := r.pkg.Scope()
	switch o := obj.(type) {
	case *types.Var:
		o = o.Origin()
		if o.IsField() {
			owner := r.fieldOwner[o]
			if owner == nil {
				return symRef{}, false
			}
			return symRef{name: owner.Name() + "." + o.Name(), kind: "field", owner: owner.Name(), member: o.Name()}, true
		}
		if o.Parent() == scope {
			return symRef{name: o.Name(), kind: "var"}, true
		}
	case *types.Func:
		o = o.Origin()
		sig, _ := o.Type().(*types.Signature)
		if sig != nil && sig.Recv() != nil {
			owner := recvNamed(sig.Recv().Type())
			kind := "method"
			if owner == nil {
				owner = r.imethOwner[o]
				kind = "imethod"
			} else if _, isIface := owner.Type().Underlying().(*types.Interface); isIface {
				kind = "imethod"
			}
			if owner == nil || owner.Pkg() != r.pkg || owner.Parent() != scope {
				return symRef{}, false
			}
			return symRef{name: owner.Name() + "." + o.Name(), kind: kind, owner: owner.Name(), member: o.Name()}, true
		}
		if o.Parent() == scope {
			return symRef{name: o.Name(), kind: "func"}, true
		}
	case *types.TypeName:
		if o.Parent() == scope {
			return symRef{name: o.Name(), kind: "type"}, true
		}
	case *types.Const:
		if o.Parent() == scope {
			return symRef{name: o.Name(), kind: "const"}, true
		}
	}
	return symRef{}, false
}

// ---------- declarations ----------

type decl struct {
	id, kind, recv, file string
	test                 bool
	loc                  int
	typeOf               string
	pkg, defPkg, how     string
}

type declRange struct {
	pos, end token.Pos
	id       string
}

func recvBaseName(e ast.Expr) string {
	for {
		switch x := e.(type) {
		case *ast.StarExpr:
			e = x.X
		case *ast.ParenExpr:
			e = x.X
		case *ast.IndexExpr:
			e = x.X
		case *ast.IndexListExpr:
			e = x.X
		case *ast.Ident:
			return x.Name
		default:
			return ""
		}
	}
}

type kv struct {
	k string
	v int
}

func sorted(m map[string]int) []kv {
	out := make([]kv, 0, len(m))
	for k, v := range m {
		out = append(out, kv{k, v})
	}
	sort.Slice(out, func(i, j int) bool {
		if out[i].v != out[j].v {
			return out[i].v > out[j].v
		}
		return out[i].k < out[j].k
	})
	return out
}

func keys(m map[string]int) []string {
	out := make([]string, 0, len(m))
	for k := range m {
		out = append(out, k)
	}
	sort.Strings(out)
	return out
}

func sum(m map[string]int) int {
	t := 0
	for _, v := range m {
		t += v
	}
	return t
}

func symExported(sym string) bool {
	if i := strings.LastIndex(sym, "."); i >= 0 {
		return ast.IsExported(sym[i+1:])
	}
	return ast.IsExported(sym)
}

type refKey struct {
	fromDecl, sym string
}

type refVal struct {
	kind  string
	to    string // target decl id
	sites int
}

func main() {
	flag.Parse()
	abs, _ := filepath.Abs(*repoFlag)
	*repoFlag = abs
	cfg := loadConfig(*confFlag)
	if err := os.MkdirAll(*outFlag, 0o755); err != nil {
		panic(err)
	}
	repF, err := os.Create(filepath.Join(*outFlag, "report.txt"))
	if err != nil {
		panic(err)
	}
	defer repF.Close()
	w := bufio.NewWriter(repF)
	defer w.Flush()
	pf := func(format string, a ...any) { fmt.Fprintf(w, format, a...) }

	all := goListJSON("./...")
	var syncPkg *goPkg
	for i := range all {
		if strings.HasSuffix(all[i].ImportPath, "/"+syncRel) {
			syncPkg = &all[i]
		}
	}
	if syncPkg == nil {
		panic("sync package not found")
	}
	syncPath := syncPkg.ImportPath
	depOut := goCmd("list", "-deps", "-test", "-tags", tags, "-f", "{{.ImportPath}}", "./...")
	depSet := map[string]bool{}
	for _, l := range strings.Split(string(depOut), "\n") {
		l = strings.TrimSpace(l)
		if l == "" || strings.Contains(l, " ") || strings.HasSuffix(l, ".test") || l == "unsafe" || l == "C" {
			continue
		}
		depSet[l] = true
	}
	var depList []string
	for d := range depSet {
		depList = append(depList, d)
	}
	sort.Strings(depList)
	expOut := goCmd(append([]string{"list", "-export", "-tags", tags, "-f", "{{.ImportPath}}\t{{.Export}}"}, depList...)...)
	exports := map[string]string{}
	for _, l := range strings.Split(string(expOut), "\n") {
		parts := strings.Split(l, "\t")
		if len(parts) == 2 && parts[1] != "" {
			exports[parts[0]] = parts[1]
		}
	}
	fset := token.NewFileSet()
	imp := importer.ForCompiler(fset, "gc", func(path string) (io.ReadCloser, error) {
		f, ok := exports[path]
		if !ok {
			return nil, fmt.Errorf("no export data for %s", path)
		}
		return os.Open(f)
	})
	parseFiles := func(dir string, names []string) []*ast.File {
		var files []*ast.File
		for _, n := range names {
			f, err := parser.ParseFile(fset, filepath.Join(dir, n), nil, parser.ParseComments|parser.SkipObjectResolution)
			if err != nil {
				fmt.Fprintf(os.Stderr, "parse %s: %v\n", n, err)
				continue
			}
			files = append(files, f)
		}
		return files
	}
	baseOf := func(p token.Pos) string { return filepath.Base(fset.Position(p).Filename) }
	lineSpan := func(a, b token.Pos) int { return fset.Position(b).Line - fset.Position(a).Line + 1 }

	prodFiles := parseFiles(syncPkg.Dir, syncPkg.GoFiles)
	testFiles := parseFiles(syncPkg.Dir, syncPkg.TestGoFiles)
	allFiles := append(append([]*ast.File{}, prodFiles...), testFiles...)
	info := &types.Info{Uses: map[*ast.Ident]types.Object{}, Defs: map[*ast.Ident]types.Object{}, Types: map[ast.Expr]types.TypeAndValue{}}
	nerr := 0
	conf := types.Config{Importer: imp, Error: func(err error) {
		nerr++
		if nerr <= 5 {
			fmt.Fprintf(os.Stderr, "typecheck: %v\n", err)
		}
	}}
	pkg, _ := conf.Check(syncPath, fset, allFiles, info)
	r := newResolver(pkg)
	scope := pkg.Scope()

	// ---- declarations ----
	decls := map[string]*decl{}
	fileRanges := map[string][]declRange{}
	isTestFile := func(f string) bool { return strings.HasSuffix(f, "_test.go") }
	addDecl := func(id, kind, recv, file string, start, end token.Pos) {
		d := decls[id]
		if d == nil {
			d = &decl{id: id, kind: kind, recv: recv, file: file, test: isTestFile(file)}
			decls[id] = d
		}
		d.loc += lineSpan(start, end)
		fileRanges[file] = append(fileRanges[file], declRange{start, end, id})
	}
	nInit := map[string]int{}
	for _, f := range allFiles {
		fname := baseOf(f.Pos())
		for _, d := range f.Decls {
			switch x := d.(type) {
			case *ast.FuncDecl:
				start := x.Pos()
				if x.Doc != nil {
					start = x.Doc.Pos()
				}
				if x.Recv != nil && len(x.Recv.List) > 0 {
					rt := recvBaseName(x.Recv.List[0].Type)
					addDecl(rt+"."+x.Name.Name, "method", rt, fname, start, x.End())
					continue
				}
				name := x.Name.Name
				if name == "init" || name == "_" {
					nInit[fname]++
					name = fmt.Sprintf("%s@%s", name, fname)
				}
				addDecl(name, "func", "", fname, start, x.End())
			case *ast.GenDecl:
				if x.Tok == token.IMPORT {
					continue
				}
				for _, sp := range x.Specs {
					switch s := sp.(type) {
					case *ast.TypeSpec:
						start := sp.Pos()
						if s.Doc != nil {
							start = s.Doc.Pos()
						} else if !x.Lparen.IsValid() && x.Doc != nil {
							start = x.Doc.Pos()
						}
						addDecl(s.Name.Name, "type", "", fname, start, sp.End())
					case *ast.ValueSpec:
						start := sp.Pos()
						if s.Doc != nil {
							start = s.Doc.Pos()
						} else if !x.Lparen.IsValid() && x.Doc != nil {
							start = x.Doc.Pos()
						}
						kind := "var"
						if x.Tok == token.CONST {
							kind = "const"
						}
						first := true
						for _, id := range s.Names {
							name := id.Name
							if name == "_" {
								name = fmt.Sprintf("_@%s:%d", fname, fset.Position(id.Pos()).Line)
							}
							if first {
								addDecl(name, kind, "", fname, start, sp.End())
								first = false
							} else {
								// further names of one spec share the range of the first
								decls[name] = &decl{id: name, kind: kind, file: fname, test: isTestFile(fname)}
							}
							if obj := info.Defs[id]; obj != nil {
								if tn := recvNamed(obj.Type()); tn != nil && tn.Pkg() == pkg && tn.Parent() == scope {
									decls[name].typeOf = tn.Name()
								}
							}
						}
					}
				}
			}
		}
	}
	for _, rs := range fileRanges {
		sort.Slice(rs, func(i, j int) bool { return rs[i].pos < rs[j].pos })
	}
	enclosing := func(file string, p token.Pos) string {
		rs := fileRanges[file]
		i := sort.Search(len(rs), func(i int) bool { return rs[i].end >= p })
		if i < len(rs) && rs[i].pos <= p {
			return rs[i].id
		}
		return ""
	}

	// ---- assignment ----
	top := cfg.famPkg["engine"]
	defPkgOfFile := func(file string) string {
		if isTestFile(file) {
			if p, ok := cfg.testFilePkg[file]; ok {
				return p
			}
			fam := family(strings.TrimSuffix(file, "_test.go") + ".go")
			if p, ok := cfg.famPkg[fam]; ok {
				return p
			}
			return top
		}
		if p, ok := cfg.famPkg[family(file)]; ok {
			return p
		}
		return top
	}
	usedConf := map[string]bool{}
	for _, d := range decls {
		d.defPkg = defPkgOfFile(d.file)
		d.pkg, d.how = d.defPkg, "default"
		if d.test {
			if p, ok := cfg.testDeclPkg[d.id]; ok {
				d.pkg, d.how = p, "testdecl"
				usedConf["testdecl "+d.id] = true
			}
			continue
		}
		if p, ok := cfg.filePkg[d.file]; ok {
			d.pkg, d.how = p, "file"
			usedConf["file "+d.file] = true
		}
		if p, ok := cfg.declPkg[d.id]; ok {
			d.pkg, d.how = p, "decl"
			usedConf["decl "+d.id] = true
		}
	}
	// typed consts follow their type; methods follow their receiver
	for _, d := range decls {
		if d.kind == "const" && d.typeOf != "" && d.how != "decl" && d.how != "testdecl" {
			// a test constant stays with its test file unless its type is test code too
			if t := decls[d.typeOf]; t != nil && t.pkg != d.pkg && (!d.test || t.test) {
				d.pkg, d.how = t.pkg, "const-follows-type"
			}
		}
	}
	for _, d := range decls {
		if d.kind != "method" {
			continue
		}
		if p, ok := cfg.toFunc[d.id]; ok {
			d.pkg, d.how = p, "tofunc"
			usedConf["tofunc "+d.id] = true
			continue
		}
		t := decls[d.recv]
		if t == nil {
			fmt.Fprintf(os.Stderr, "method %s: receiver type not found\n", d.id)
			continue
		}
		if t.pkg != d.pkg {
			d.pkg, d.how = t.pkg, "method-follows-type"
		}
	}
	for k := range cfg.testDeclPkg {
		if !usedConf["testdecl "+k] {
			fmt.Fprintf(os.Stderr, "conf: testdecl %s matched nothing\n", k)
		}
	}
	for p, m := range cfg.testCopy {
		for k := range m {
			if d := decls[k]; d == nil || !d.test {
				fmt.Fprintf(os.Stderr, "conf: testcopy %s %s is not a test declaration\n", p, k)
			}
		}
	}
	for k := range cfg.filePkg {
		if !usedConf["file "+k] {
			fmt.Fprintf(os.Stderr, "conf: file %s matched nothing\n", k)
		}
	}
	for k := range cfg.declPkg {
		if !usedConf["decl "+k] {
			fmt.Fprintf(os.Stderr, "conf: decl %s matched nothing\n", k)
		}
	}
	for k := range cfg.toFunc {
		if !usedConf["tofunc "+k] {
			fmt.Fprintf(os.Stderr, "conf: tofunc %s matched nothing\n", k)
		}
	}

	// allowed-import closure
	reach := map[string]map[string]bool{}
	var visit func(root, p string)
	visit = func(root, p string) {
		for q := range cfg.allow[p] {
			if !reach[root][q] {
				reach[root][q] = true
				visit(root, q)
			}
		}
	}
	for _, p := range cfg.order {
		reach[p] = map[string]bool{}
		visit(p, p)
	}

	// ---- references ----
	refs := map[refKey]*refVal{}
	for id, obj := range info.Uses {
		ref, ok := r.resolve(obj)
		if !ok {
			continue
		}
		file := baseOf(id.Pos())
		from := enclosing(file, id.Pos())
		if from == "" {
			continue
		}
		to := ref.name
		if ref.kind == "field" || ref.kind == "imethod" {
			to = ref.owner
		} else if ref.kind == "method" {
			if decls[to] == nil {
				to = ref.owner
			}
		}
		if decls[to] == nil {
			continue
		}
		k := refKey{from, ref.name}
		if refs[k] == nil {
			refs[k] = &refVal{kind: ref.kind, to: to}
		}
		refs[k].sites++
	}

	// ---- production package graph ----
	type edge struct{ a, b string }
	edgeSyms := map[edge]map[string]int{}
	symKind := map[string]string{}
	symPkg := map[string]string{}
	symUsersFrom := map[string]map[string]int{} // sym -> from pkg -> sites (cross-package, production)
	type viol struct {
		fromDecl, fromFile, fromPkg, sym, toPkg string
		sites                                   int
	}
	var viols []viol
	// consumer-side interfaces: consumer package -> concrete type -> method -> sites
	ifaceUse := map[string]map[string]map[string]int{}
	ifaceTypeSites := map[string]map[string]int{}
	for k, v := range refs {
		fd, td := decls[k.fromDecl], decls[v.to]
		symKind[k.sym] = v.kind
		symPkg[k.sym] = td.pkg
		if fd.test || td.test {
			continue
		}
		if fd.pkg == td.pkg {
			continue
		}
		// a type the consumer sees only through an interface it declares itself:
		// naming the type and calling its exported methods creates no import
		if via := cfg.viaIface[fd.pkg]; via != nil {
			owner, member := k.sym, ""
			if i := strings.Index(k.sym, "."); i >= 0 {
				owner, member = k.sym[:i], k.sym[i+1:]
			}
			if via[owner] && (v.kind == "type" || (v.kind == "method" && ast.IsExported(member))) {
				if ifaceUse[fd.pkg] == nil {
					ifaceUse[fd.pkg] = map[string]map[string]int{}
					ifaceTypeSites[fd.pkg] = map[string]int{}
				}
				if ifaceUse[fd.pkg][owner] == nil {
					ifaceUse[fd.pkg][owner] = map[string]int{}
				}
				if member == "" {
					ifaceTypeSites[fd.pkg][owner] += v.sites
				} else {
					ifaceUse[fd.pkg][owner][member] += v.sites
				}
				continue
			}
		}
		e := edge{fd.pkg, td.pkg}
		if edgeSyms[e] == nil {
			edgeSyms[e] = map[string]int{}
		}
		edgeSyms[e][k.sym] += v.sites
		if symUsersFrom[k.sym] == nil {
			symUsersFrom[k.sym] = map[string]int{}
		}
		symUsersFrom[k.sym][fd.pkg] += v.sites
		if !cfg.allow[fd.pkg][td.pkg] {
			viols = append(viols, viol{k.fromDecl, fd.file, fd.pkg, k.sym, td.pkg, v.sites})
		}
	}

	// the complete cross-package reference ledger, one row per (using declaration, symbol)
	{
		rf, _ := os.Create(filepath.Join(*outFlag, "refs.tsv"))
		rw := bufio.NewWriter(rf)
		fmt.Fprintf(rw, "from_pkg\tfrom_decl\tfrom_file\tfrom_is_test\tto_pkg\tto_decl\tto_is_test\tsymbol\tkind\texported_today\tsites\n")
		type refRow struct {
			k refKey
			v *refVal
		}
		var rr []refRow
		for k, v := range refs {
			if decls[k.fromDecl].pkg != decls[v.to].pkg {
				rr = append(rr, refRow{k, v})
			}
		}
		sort.Slice(rr, func(i, j int) bool {
			a, b := rr[i], rr[j]
			ap, bp := decls[a.k.fromDecl].pkg+"\x00"+decls[a.v.to].pkg, decls[b.k.fromDecl].pkg+"\x00"+decls[b.v.to].pkg
			if ap != bp {
				return ap < bp
			}
			if a.k.sym != b.k.sym {
				return a.k.sym < b.k.sym
			}
			return a.k.fromDecl < b.k.fromDecl
		})
		for _, x := range rr {
			fd, td := decls[x.k.fromDecl], decls[x.v.to]
			fmt.Fprintf(rw, "%s\t%s\t%s\t%v\t%s\t%s\t%v\t%s\t%s\t%v\t%d\n", fd.pkg, x.k.fromDecl, fd.file, fd.test, td.pkg, x.v.to, td.test,
				x.k.sym, x.v.kind, symExported(x.k.sym), x.v.sites)
		}
		rw.Flush()
		rf.Close()
	}

	// every method of a consumer-side interface must have a signature the
	// consumer can spell: only types from packages it may import
	type ifaceLine struct {
		consumer, typ, method, sig string
		sites                      int
		bad                        []string
	}
	var ifaceLines []ifaceLine
	nIfaceBad := 0
	for _, consumer := range cfg.order {
		for _, typ := range keys(boolInt(cfg.viaIface[consumer])) {
			tn, _ := scope.Lookup(typ).(*types.TypeName)
			if tn == nil {
				fmt.Fprintf(os.Stderr, "conf: viaiface %s %s: no such type\n", consumer, typ)
				continue
			}
			for _, m := range keys(ifaceUse[consumer][typ]) {
				obj, _, _ := types.LookupFieldOrMethod(types.NewPointer(tn.Type()), true, pkg, m)
				fn, _ := obj.(*types.Func)
				line := ifaceLine{consumer: consumer, typ: typ, method: m, sites: ifaceUse[consumer][typ][m]}
				if fn == nil {
					line.bad = append(line.bad, "method not found")
				} else {
					line.sig = types.TypeString(fn.Type(), func(p *types.Package) string {
						if p == pkg {
							return ""
						}
						return p.Name()
					})
					walkNamed(fn.Type(), pkg, map[types.Type]bool{}, func(n *types.TypeName) {
						d := decls[n.Name()]
						if d == nil {
							return
						}
						if d.pkg != consumer && !reach[consumer][d.pkg] {
							line.bad = append(line.bad, fmt.Sprintf("%s is in %s", n.Name(), d.pkg))
						}
					})
				}
				nIfaceBad += len(line.bad)
				ifaceLines = append(ifaceLines, line)
			}
		}
	}

	pf("# endstate report  conf=%s  package=%s  typecheck errors=%d\n", *confFlag, syncPath, nerr)
	pf("# build-excluded files (not type-checked): %v\n\n", syncPkg.IgnoredGoFiles)

	// sizes
	pf("## 1. packages in the end state (production declarations)\n")
	pkgDecls := map[string]int{}
	pkgLOC := map[string]int{}
	pkgFiles := map[string]map[string]int{}
	fileSplit := map[string]map[string]int{}
	for _, d := range decls {
		if d.test {
			continue
		}
		pkgDecls[d.pkg]++
		pkgLOC[d.pkg] += d.loc
		if pkgFiles[d.pkg] == nil {
			pkgFiles[d.pkg] = map[string]int{}
		}
		pkgFiles[d.pkg][d.file] += d.loc
		if fileSplit[d.file] == nil {
			fileSplit[d.file] = map[string]int{}
		}
		fileSplit[d.file][d.pkg] += d.loc
	}
	pf("%-14s %6s %8s %7s   allowed imports\n", "package", "decls", "declLOC", "files")
	for _, p := range cfg.order {
		pf("%-14s %6d %8d %7d   %s\n", p, pkgDecls[p], pkgLOC[p], len(pkgFiles[p]), strings.Join(keys(boolInt(cfg.allow[p])), " "))
	}
	for p := range pkgDecls {
		if reach[p] == nil {
			pf("!! package %s is assigned but missing from 'order'\n", p)
		}
	}

	pf("\n## 2. package dependency matrix (row imports column): distinct symbols / reference sites; '!' = not an allowed import\n")
	pf("%-14s", "A\\B")
	for _, b := range cfg.order {
		pf(" %14s", b)
	}
	pf("\n")
	for _, a := range cfg.order {
		pf("%-14s", a)
		for _, b := range cfg.order {
			m := edgeSyms[edge{a, b}]
			cell := "."
			if len(m) > 0 {
				cell = fmt.Sprintf("%d/%d", len(m), sum(m))
				if !cfg.allow[a][b] {
					cell += "!"
				}
			}
			pf(" %14s", cell)
		}
		pf("\n")
	}
	// cycle check on the actual graph
	adj := map[string][]string{}
	for e := range edgeSyms {
		adj[e.a] = append(adj[e.a], e.b)
	}
	state := map[string]int{}
	var cyc []string
	var stack []string
	var dfs func(n string)
	dfs = func(n string) {
		state[n] = 1
		stack = append(stack, n)
		for _, m := range adj[n] {
			if state[m] == 1 {
				i := len(stack) - 1
				for i >= 0 && stack[i] != m {
					i--
				}
				cyc = append(cyc, strings.Join(append(append([]string{}, stack[i:]...), m), " -> "))
			} else if state[m] == 0 {
				dfs(m)
			}
		}
		stack = stack[:len(stack)-1]
		state[n] = 2
	}
	for _, p := range cfg.order {
		if state[p] == 0 {
			dfs(p)
		}
	}
	sort.Strings(cyc)
	if len(cyc) == 0 {
		pf("\nimport graph: ACYCLIC\n")
	} else {
		pf("\nimport graph: %d CYCLE(S)\n", len(cyc))
		for _, c := range cyc {
			pf("  %s\n", c)
		}
	}

	// violations
	sort.Slice(viols, func(i, j int) bool {
		a, b := viols[i], viols[j]
		if a.fromPkg != b.fromPkg {
			return a.fromPkg < b.fromPkg
		}
		if a.toPkg != b.toPkg {
			return a.toPkg < b.toPkg
		}
		if a.sym != b.sym {
			return a.sym < b.sym
		}
		return a.fromDecl < b.fromDecl
	})
	pf("\n## 3. references outside the allowed imports: %d (decl -> symbol)\n", len(viols))
	vf, _ := os.Create(filepath.Join(*outFlag, "violations.tsv"))
	vw := bufio.NewWriter(vf)
	fmt.Fprintf(vw, "from_pkg\tto_pkg\tfrom_decl\tfrom_file\tfrom_how\tsymbol\tkind\tsym_file\tsites\n")
	lastPair := ""
	bySym := map[string]map[string]int{}
	for _, v := range viols {
		pair := v.fromPkg + " -> " + v.toPkg
		if bySym[pair] == nil {
			bySym[pair] = map[string]int{}
		}
		bySym[pair][v.sym] += v.sites
		td := decls[refs[refKey{v.fromDecl, v.sym}].to]
		fmt.Fprintf(vw, "%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%d\n", v.fromPkg, v.toPkg, v.fromDecl, v.fromFile, decls[v.fromDecl].how, v.sym, symKind[v.sym], td.file, v.sites)
		_ = lastPair
	}
	vw.Flush()
	vf.Close()
	var pairs []string
	for p := range bySym {
		pairs = append(pairs, p)
	}
	sort.Strings(pairs)
	for _, p := range pairs {
		pf("\n### %s : %d symbols / %d sites\n", p, len(bySym[p]), sum(bySym[p]))
		for _, e := range sorted(bySym[p]) {
			// who references it
			var from []string
			for _, v := range viols {
				if v.fromPkg+" -> "+v.toPkg == p && v.sym == e.k {
					from = append(from, v.fromDecl)
				}
			}
			if len(from) > 6 {
				from = append(from[:6], fmt.Sprintf("...+%d", len(from)-6))
			}
			td := symDecl(decls, e.k)
			file := "?"
			if td != nil {
				file = td.file
			}
			pf("  %-50s %-7s %-32s %3d  <- %s\n", e.k, symKind[e.k], file, e.v, strings.Join(from, ", "))
		}
	}

	pf("\n## 3b. consumer-side interfaces (the consumer names the concrete type today; in the end state it declares an interface with these methods): %d problems\n", nIfaceBad)
	for _, consumer := range cfg.order {
		for _, typ := range keys(boolInt(cfg.viaIface[consumer])) {
			pf("  %s sees %s (%s) through its own interface; the type is named at %d sites\n", consumer, typ, decls[typ].pkg, ifaceTypeSites[consumer][typ])
			for _, l := range ifaceLines {
				if l.consumer != consumer || l.typ != typ {
					continue
				}
				mark := "ok"
				if len(l.bad) > 0 {
					mark = "NOT SPELLABLE: " + strings.Join(l.bad, "; ")
				}
				pf("    %-28s %2d sites  %s  [%s]\n", l.method, l.sites, l.sig, mark)
			}
		}
	}

	// ---- moves ----
	pf("\n## 4. declarations whose package differs from their file's default (production)\n")
	mf, _ := os.Create(filepath.Join(*outFlag, "moves.tsv"))
	mw := bufio.NewWriter(mf)
	fmt.Fprintf(mw, "decl\tkind\tfile\tdefault_pkg\tfinal_pkg\thow\tloc\n")
	type mv struct{ from, to, how string }
	mvCount := map[mv]int{}
	mvLOC := map[mv]int{}
	var ids []string
	for id := range decls {
		ids = append(ids, id)
	}
	sort.Strings(ids)
	for _, id := range ids {
		d := decls[id]
		if d.test || d.pkg == d.defPkg {
			continue
		}
		fmt.Fprintf(mw, "%s\t%s\t%s\t%s\t%s\t%s\t%d\n", d.id, d.kind, d.file, d.defPkg, d.pkg, d.how, d.loc)
		k := mv{d.defPkg, d.pkg, d.how}
		mvCount[k]++
		mvLOC[k] += d.loc
	}
	mw.Flush()
	mf.Close()
	var mvs []mv
	for k := range mvCount {
		mvs = append(mvs, k)
	}
	sort.Slice(mvs, func(i, j int) bool {
		if mvs[i].from != mvs[j].from {
			return mvs[i].from < mvs[j].from
		}
		if mvs[i].to != mvs[j].to {
			return mvs[i].to < mvs[j].to
		}
		return mvs[i].how < mvs[j].how
	})
	for _, k := range mvs {
		pf("  %-14s -> %-14s %-20s %4d decls %6d LOC\n", k.from, k.to, k.how, mvCount[k], mvLOC[k])
	}
	pf("\nfiles whose declarations end up in more than one package (file must be split):\n")
	var fnames []string
	for f := range fileSplit {
		fnames = append(fnames, f)
	}
	sort.Strings(fnames)
	for _, f := range fnames {
		if len(fileSplit[f]) > 1 {
			pf("  %-36s", f)
			for _, e := range sorted(fileSplit[f]) {
				pf(" %s=%d", e.k, e.v)
			}
			pf("\n")
		}
	}
	pf("\nfiles that move whole to a package other than their family default:\n")
	for _, f := range fnames {
		if len(fileSplit[f]) == 1 {
			for p := range fileSplit[f] {
				if p != defPkgOfFile(f) {
					pf("  %-36s %s -> %s (%d LOC)\n", f, defPkgOfFile(f), p, fileSplit[f][p])
				}
			}
		}
	}

	// ---- export ledger ----
	pf("\n## 5. export ledger: unexported identifiers referenced from another package (production code)\n")
	ef, _ := os.Create(filepath.Join(*outFlag, "edges.tsv"))
	ew := bufio.NewWriter(ef)
	fmt.Fprintf(ew, "from_pkg\tto_pkg\tsymbol\tkind\texported\tsym_file\tsites\n")
	needExport := map[string]bool{}
	expByPkgKind := map[string]map[string]int{}
	alreadyByPkg := map[string]int{}
	for _, a := range cfg.order {
		for _, b := range cfg.order {
			m := edgeSyms[edge{a, b}]
			for _, e := range sorted(m) {
				td := symDecl(decls, e.k)
				file := "?"
				if td != nil {
					file = td.file
				}
				ex := symExported(e.k)
				fmt.Fprintf(ew, "%s\t%s\t%s\t%s\t%v\t%s\t%d\n", a, b, e.k, symKind[e.k], ex, file, e.v)
				if !ex && !needExport[e.k] {
					needExport[e.k] = true
					if expByPkgKind[b] == nil {
						expByPkgKind[b] = map[string]int{}
					}
					expByPkgKind[b][symKind[e.k]]++
				}
			}
		}
	}
	ew.Flush()
	ef.Close()
	surface := map[string]map[string]bool{}
	for e, m := range edgeSyms {
		if surface[e.b] == nil {
			surface[e.b] = map[string]bool{}
		}
		for s := range m {
			surface[e.b][s] = true
			if symExported(s) {
				alreadyByPkg[e.b]++
			}
		}
	}
	for p, m := range surface {
		n := 0
		for s := range m {
			if symExported(s) {
				n++
			}
		}
		alreadyByPkg[p] = n
	}
	pf("%-14s %9s %9s   to export, by kind\n", "package", "surface", "exported")
	for _, p := range cfg.order {
		pf("%-14s %9d %9d   %v\n", p, len(surface[p]), alreadyByPkg[p], expByPkgKind[p])
	}
	// owner types unexported while a member is referenced
	xf, _ := os.Create(filepath.Join(*outFlag, "exports.tsv"))
	xw := bufio.NewWriter(xf)
	fmt.Fprintf(xw, "pkg\tsymbol\tkind\tsym_file\tused_from\tsites\n")
	var need []string
	for s := range needExport {
		need = append(need, s)
	}
	sort.Slice(need, func(i, j int) bool {
		if symPkg[need[i]] != symPkg[need[j]] {
			return symPkg[need[i]] < symPkg[need[j]]
		}
		return need[i] < need[j]
	})
	for _, s := range need {
		td := symDecl(decls, s)
		file := "?"
		if td != nil {
			file = td.file
		}
		var from []string
		for _, e := range sorted(symUsersFrom[s]) {
			from = append(from, fmt.Sprintf("%s=%d", e.k, e.v))
		}
		fmt.Fprintf(xw, "%s\t%s\t%s\t%s\t%s\t%d\n", symPkg[s], s, symKind[s], file, strings.Join(from, " "), sum(symUsersFrom[s]))
	}
	xw.Flush()
	xf.Close()

	// unexported field reach per struct
	pf("\nunexported struct fields reached from another package (struct: fields/sites, by using package):\n")
	type fr struct {
		fields map[string]bool
		by     map[string]int
	}
	frs := map[string]*fr{}
	for s := range needExport {
		if symKind[s] != "field" {
			continue
		}
		owner := s[:strings.Index(s, ".")]
		if frs[owner] == nil {
			frs[owner] = &fr{fields: map[string]bool{}, by: map[string]int{}}
		}
		frs[owner].fields[s] = true
		for p, c := range symUsersFrom[s] {
			frs[owner].by[p] += c
		}
	}
	var owners []string
	for o := range frs {
		owners = append(owners, o)
	}
	sort.Slice(owners, func(i, j int) bool { return sum(frs[owners[i]].by) > sum(frs[owners[j]].by) })
	totF, totS := 0, 0
	for _, o := range owners {
		totF += len(frs[o].fields)
		totS += sum(frs[o].by)
		var by []string
		for _, e := range sorted(frs[o].by) {
			by = append(by, fmt.Sprintf("%s=%d", e.k, e.v))
		}
		pf("  %-34s %-13s %3d fields %4d sites  %s\n", o, decls[o].pkg, len(frs[o].fields), sum(frs[o].by), strings.Join(by, " "))
	}
	pf("  TOTAL %d structs, %d fields, %d sites\n", len(owners), totF, totS)

	// package-level vars crossing packages
	pf("\npackage-level variables referenced from another package:\n")
	for _, a := range cfg.order {
		for _, b := range cfg.order {
			for _, e := range sorted(edgeSyms[edge{a, b}]) {
				if symKind[e.k] == "var" {
					pf("  %-14s -> %-14s %-44s %s %d sites\n", a, b, e.k, symDecl(decls, e.k).file, e.v)
				}
			}
		}
	}

	// interfaces with unexported methods implemented in another package
	pf("\ninterfaces with unexported methods and an implementing type in another package:\n")
	var typeNames []*types.TypeName
	for _, n := range scope.Names() {
		if tn, ok := scope.Lookup(n).(*types.TypeName); ok && !tn.IsAlias() {
			typeNames = append(typeNames, tn)
		}
	}
	nIfc := 0
	ifaceMethodExport := map[string]bool{} // interface methods that must be exported so another package can implement them
	for _, in := range typeNames {
		id := decls[in.Name()]
		if id == nil || id.test {
			continue
		}
		iface, ok := in.Type().Underlying().(*types.Interface)
		if !ok || iface.NumMethods() == 0 {
			continue
		}
		var unexp []string
		for i := 0; i < iface.NumMethods(); i++ {
			if !iface.Method(i).Exported() {
				unexp = append(unexp, iface.Method(i).Name())
			}
		}
		if len(unexp) == 0 {
			continue
		}
		for _, tn := range typeNames {
			td := decls[tn.Name()]
			if td == nil || tn == in {
				continue
			}
			if _, isI := tn.Type().Underlying().(*types.Interface); isI {
				continue
			}
			if td.pkg == id.pkg {
				continue
			}
			if types.Implements(tn.Type(), iface) || types.Implements(types.NewPointer(tn.Type()), iface) {
				what := "production"
				if td.test {
					what = "test"
				}
				pf("  interface %s (%s, %s) methods %v  <- implemented by %s type %s (%s, %s)\n", in.Name(), id.pkg, id.file, unexp, what, tn.Name(), td.pkg, td.file)
				for _, m := range unexp {
					ifaceMethodExport[in.Name()+"."+m] = true
					ifaceMethodExport[tn.Name()+"."+m] = true // the implementing method is renamed with it
				}
				nIfc++
			}
		}
	}
	if nIfc == 0 {
		pf("  none\n")
	}

	// unkeyed composite literals of a struct from another package
	pf("\nunkeyed composite literals of a struct that ends up in another package:\n")
	nLit := 0
	for _, f := range prodFiles {
		fname := baseOf(f.Pos())
		ast.Inspect(f, func(n ast.Node) bool {
			cl, ok := n.(*ast.CompositeLit)
			if !ok || len(cl.Elts) == 0 {
				return true
			}
			if _, kvp := cl.Elts[0].(*ast.KeyValueExpr); kvp {
				return true
			}
			tv, ok := info.Types[cl]
			if !ok {
				return true
			}
			tn := recvNamed(tv.Type)
			if tn == nil || tn.Pkg() != pkg {
				return true
			}
			if _, isStruct := tn.Type().Underlying().(*types.Struct); !isStruct {
				return true
			}
			from := enclosing(fname, cl.Pos())
			if from == "" || decls[tn.Name()] == nil {
				return true
			}
			if decls[from].pkg != decls[tn.Name()].pkg {
				pf("  %s:%d %s literal in %s (%s)\n", fname, fset.Position(cl.Pos()).Line, tn.Name(), from, decls[from].pkg)
				nLit++
			}
			return true
		})
	}
	if nLit == 0 {
		pf("  none\n")
	}
	pf("\ninit functions: %v\n", nInit)

	// const blocks split across packages
	pf("\nconst blocks whose constants end up in more than one package (a block using iota or implicit repetition cannot be split blindly):\n")
	nSplitConst := 0
	for _, f := range prodFiles {
		fname := baseOf(f.Pos())
		for _, d := range f.Decls {
			gd, ok := d.(*ast.GenDecl)
			if !ok || gd.Tok != token.CONST || !gd.Lparen.IsValid() {
				continue
			}
			pk := map[string]int{}
			implicit, usesIota := false, false
			for _, sp := range gd.Specs {
				vs := sp.(*ast.ValueSpec)
				if len(vs.Values) == 0 {
					implicit = true
				}
				for _, v := range vs.Values {
					ast.Inspect(v, func(n ast.Node) bool {
						if id, ok := n.(*ast.Ident); ok && id.Name == "iota" {
							usesIota = true
						}
						return true
					})
				}
				for _, n := range vs.Names {
					if dd := decls[n.Name]; dd != nil {
						pk[dd.pkg]++
					}
				}
			}
			if len(pk) > 1 {
				nSplitConst++
				pf("  %s:%d  %v  iota=%v implicit-repeat=%v\n", fname, fset.Position(gd.Pos()).Line, pk, usesIota, implicit)
			}
		}
	}
	if nSplitConst == 0 {
		pf("  none\n")
	}

	// exported-name collisions
	nColl := 0
	checkCollision := func(s string) {
		nn := exportedName(s)
		if nn == s {
			return
		}
		if i := strings.Index(s, "."); i >= 0 {
			tn, _ := scope.Lookup(s[:i]).(*types.TypeName)
			if tn == nil {
				return
			}
			for _, t := range []types.Type{tn.Type(), types.NewPointer(tn.Type())} {
				if obj, _, _ := types.LookupFieldOrMethod(t, true, pkg, nn[i+1:]); obj != nil {
					pf("  %s -> %s already exists on the type\n", s, nn)
					nColl++
					return
				}
			}
			return
		}
		if d := decls[nn]; d != nil && d.pkg == symPkg[s] {
			pf("  %s -> %s already exists in %s (%s)\n", s, nn, d.pkg, d.file)
			nColl++
		}
	}
	pf("\nexported-name collisions (identifier to export whose capitalised name already exists in its end-state package):\n")
	for _, s := range need {
		checkCollision(s)
	}
	for _, s := range keys(boolInt(ifaceMethodExport)) {
		if !needExport[s] {
			checkCollision(s)
		}
	}
	for _, s := range keys(boolInt(cfg.testExport)) {
		if !needExport[s] && !ifaceMethodExport[s] {
			checkCollision(s)
		}
	}
	if nColl == 0 {
		pf("  none\n")
	}

			continue
		}
		from := enclosing(baseOf(id.Pos()), id.Pos())
		if from == "" || decls[from] == nil {
			continue
		}
		d := decls[from]
		if extImp[d.test][d.pkg] == nil {
			extImp[d.test][d.pkg] = map[string]int{}
		}
		extImp[d.test][d.pkg][pn.Imported().Path()]++
	}
	pf("\nimports from outside internal/sync, per end-state package (production code; sites):\n")
	xi, _ := os.Create(filepath.Join(*outFlag, "extimports.tsv"))
	xiw := bufio.NewWriter(xi)
	fmt.Fprintf(xiw, "pkg\ttest\timport\tsites\n")
	for _, isTest := range []bool{false, true} {
		for _, p := range cfg.order {
			m := extImp[isTest][p]
			var internal, other []string
			for _, e := range sorted(m) {
				fmt.Fprintf(xiw, "%s\t%v\t%s\t%d\n", p, isTest, e.k, e.v)
				if strings.HasPrefix(e.k, modPrefix) {
					internal = append(internal, fmt.Sprintf("%s=%d", strings.TrimPrefix(e.k, modPrefix), e.v))
				} else if strings.Contains(e.k, ".") || e.k == "database/sql" || e.k == "net/http" || e.k == "os" {
					other = append(other, fmt.Sprintf("%s=%d", e.k, e.v))
				}
			}
			if isTest {
				continue
			}
			pf("  %-14s module: %s\n", p, strings.Join(internal, " "))
			pf("  %-14s other:  %s\n", "", strings.Join(other, " "))
		}
	}
	xiw.Flush()
	xi.Close()
	pf("\nblank (side-effect) imports:\n")
	for _, f := range allFiles {
		for _, is := range f.Imports {
			if is.Name != nil && is.Name.Name == "_" {
				pf("  %s imports %s\n", baseOf(f.Pos()), is.Path.Value)
			}
		}
	}

	// ---- raw dumps for offline iteration of the test configuration ----
	{
		df, _ := os.Create(filepath.Join(*outFlag, "rawdecls.tsv"))
		dw := bufio.NewWriter(df)
		fmt.Fprintf(dw, "id\tkind\trecv\tfile\ttest\tloc\ttypeof\tpkg\thow\n")
		var ids []string
		for id := range decls {
			ids = append(ids, id)
		}
		sort.Strings(ids)
		for _, id := range ids {
			d := decls[id]
			fmt.Fprintf(dw, "%s\t%s\t%s\t%s\t%v\t%d\t%s\t%s\t%s\n", d.id, d.kind, d.recv, d.file, d.test, d.loc, d.typeOf, d.pkg, d.how)
		}
		dw.Flush()
		df.Close()
		rf, _ := os.Create(filepath.Join(*outFlag, "rawrefs.tsv"))
		rw := bufio.NewWriter(rf)
		fmt.Fprintf(rw, "from_decl\tto_decl\tsymbol\tkind\tsites\texported_end\n")
		type rr struct {
			k refKey
			v *refVal
		}
		var all []rr
		for k, v := range refs {
			if decls[k.fromDecl].test {
				all = append(all, rr{k, v})
			}
		}
		sort.Slice(all, func(i, j int) bool {
			if all[i].k.fromDecl != all[j].k.fromDecl {
				return all[i].k.fromDecl < all[j].k.fromDecl
			}
			return all[i].k.sym < all[j].k.sym
		})
		for _, x := range all {
			fmt.Fprintf(rw, "%s\t%s\t%s\t%s\t%d\t%v\n", x.k.fromDecl, x.v.to, x.k.sym, x.v.kind, x.v.sites,
				symExported(x.k.sym) || needExport[x.k.sym] || ifaceMethodExport[x.k.sym])
		}
		rw.Flush()
		rf.Close()

	// ---- tests ----
	pf("\n## 6. tests (%d files, all in package sync today)\n", len(syncPkg.TestGoFiles))
	exportedEnd := func(s string) bool {
		return symExported(s) || needExport[s] || ifaceMethodExport[s] || cfg.testExport[s]
	}
	copyRoot := func(d *decl) string {
		switch {
		case d.kind == "method":
			return d.recv
		case d.kind == "const" && d.typeOf != "":
			return d.typeOf
		}
		return d.id
	}
	type tviol struct {
		class, fromPkg, toPkg, fromDecl, fromFile, sym string
		sites                                          int
	}
	var tviols []tviol
	testEdge := map[edge]map[string]int{}       // tests (or helper code) of a -> package b: symbol -> sites
	helperExport := map[string]map[string]int{} // helper symbol that must be exported -> user package -> sites
	helperUse := map[string]map[string]int{}    // helper declaration -> user package -> sites (cross-package)
	seamUse := map[string]map[string]int{}      // testexport symbol -> user package -> sites
	fileViol := map[string]map[string]int{}     // test file -> class -> references
	fileUnexp := map[string]map[string]map[string]bool{}
	bump := func(m map[string]map[string]int, a, b string, n int) {
		if m[a] == nil {
			m[a] = map[string]int{}
		}
		m[a][b] += n
	}
	checkTestRef := func(p string, k refKey, v *refVal) {
		fd, td := decls[k.fromDecl], decls[v.to]
		q := td.pkg
		if p == q {
			return
		}
		if td.test && (cfg.testCopy[p][td.id] || cfg.testCopy[p][copyRoot(td)]) {
			return // the tests of p keep their own copy of this helper
		}
		class := ""
		switch {
		case td.test && !cfg.helperPkg[q]:
			class = "other-package-test-code"
		case cfg.helperPkg[p] && !cfg.allow[p][q]:
			class = "helper-import-not-allowed"
		case !cfg.helperPkg[p] && reach[q][p]:
			class = "import-cycle"
		case !exportedEnd(k.sym) && !td.test:
			class = "unexported-production-symbol"
		}
		if class != "" {
			tviols = append(tviols, tviol{class, p, q, k.fromDecl, fd.file, k.sym, v.sites})
			bump(fileViol, fd.file, class, 1)
			return
		}
		e := edge{p, q}
		if testEdge[e] == nil {
			testEdge[e] = map[string]int{}
		}
		testEdge[e][k.sym] += v.sites
		if td.test {
			bump(helperUse, v.to, p, v.sites)
			if !symExported(k.sym) {
				bump(helperExport, k.sym, p, v.sites)
			}
		} else if !symExported(k.sym) && !needExport[k.sym] && !ifaceMethodExport[k.sym] {
			bump(seamUse, k.sym, p, v.sites)
		}
	}
	for k, v := range refs {
		fd, td := decls[k.fromDecl], decls[v.to]
		if !fd.test {
			continue
		}
		checkTestRef(fd.pkg, k, v)
		for p, m := range cfg.testCopy {
			if p != fd.pkg && (m[fd.id] || m[copyRoot(fd)]) {
				checkTestRef(p, k, v)
			}
		}
		if !td.test && !symExported(k.sym) && !needExport[k.sym] && !ifaceMethodExport[k.sym] {
			if fileUnexp[fd.file] == nil {
				fileUnexp[fd.file] = map[string]map[string]bool{}
			}
			if fileUnexp[fd.file][td.pkg] == nil {
				fileUnexp[fd.file][td.pkg] = map[string]bool{}
			}
			fileUnexp[fd.file][td.pkg][k.sym] = true
		}
	}

	// where test code lives

	testDecls := map[string]int{}
	homeFiles := map[string]int{}
	for _, d := range decls {

		}

		tstats[d.file].decls[d.pkg]++
		testLOC[d.pkg] += d.loc

		if cfg.helperPkg[d.pkg] && d.kind == "func" && hasAnyPrefix(d.id, "Test", "Benchmark", "Fuzz", "Example") {
			pf("!! %s (%s) is a test function assigned to helper package %s\n", d.id, d.file, d.pkg)
		}
	}
	var tnames []string
	for f, t := range tstats {
		tnames = append(tnames, f)
		homeFiles[t.home]++
	}
	sort.Strings(tnames)
	pf("where test code lives (a helper package is an ordinary package; everything else is in-package _test.go files):\n")
	pf("  %-16s %6s %6s %8s  %s\n", "package", "files", "decls", "declLOC", "may be imported by the tests of")
	for _, p := range cfg.order {
		kind := ""
		if cfg.helperPkg[p] {
			kind = " (helper; imports " + strings.Join(keys(boolInt(cfg.allow[p])), " ") + ")"
		}
		var users []string
		for _, q := range cfg.order {
			if q != p && !cfg.helperPkg[q] && !reach[p][q] {
				users = append(users, q)
			}
		}
		pf("  %-16s %6d %6d %8d  %s%s\n", p, homeFiles[p], testDecls[p], testLOC[p], strings.Join(users, " "), kind)
	}

	pf("\ntest import matrix (tests of row, or helper row, import column): distinct symbols / sites\n")
	pf("%-16s", "A\\B")
	for _, b := range cfg.order {

	pf("\n")
	for _, a := range cfg.order {

		for _, b := range cfg.order {

					loc += d.loc
				}
				for _, d := range decls {
					if d.test && d.kind == "method" && d.recv == h {
						loc += d.loc
					}
				}
			}
			pf("  %-16s %s  (%d LOC duplicated)\n", p, strings.Join(keys(boolInt(cfg.testCopy[p])), " "), loc)
		}
	}

	tf, _ := os.Create(filepath.Join(*outFlag, "tests.tsv"))
	tw := bufio.NewWriter(tf)
	fmt.Fprintf(tw, "test_file\thome\tloc_by_pkg\tviolations\tunexported_used_by_pkg\n")
	nMovedFiles, nSplitFiles := 0, 0
	for _, f := range tnames {
		t := tstats[f]
		var by, vs, un []string
		for _, e := range sorted(t.byPkg) {
			by = append(by, fmt.Sprintf("%s=%d", e.k, e.v))
		}
		for _, e := range sorted(fileViol[f]) {
			vs = append(vs, fmt.Sprintf("%s=%d", e.k, e.v))
		}
		for p, m := range fileUnexp[f] {
			un = append(un, fmt.Sprintf("%s=%d", p, len(m)))
		}
		sort.Strings(un)
		if len(t.byPkg) > 1 {
			nSplitFiles++
		}
		if family(strings.TrimSuffix(f, "_test.go")+".go") != "" && cfg.testFilePkg[f] != "" {
			nMovedFiles++
		}
		fmt.Fprintf(tw, "%s\t%s\t%s\t%s\t%s\n", f, t.home, strings.Join(by, " "), strings.Join(vs, " "), strings.Join(un, " "))
	}
	tw.Flush()
	tf.Close()
	pf("\ntest files with an explicit home (testfile): %d; test files whose declarations end up in more than one package: %d\n", nMovedFiles, nSplitFiles)

	// per-declaration view of test code: what each root declaration needs and who uses it
	{
		type rootInfo struct {
			loc        int
			needsProd  map[string]map[string]bool // package -> symbols
			needsUnexp map[string]bool            // pkg:symbol, not exported in the end state
			needsTest  map[string]bool            // other root test declarations
			usedBy     map[string]int             // package -> sites (from other roots)
			usedFiles  map[string]bool
		}
		roots := map[string]*rootInfo{}
		get := func(id string) *rootInfo {
			r := roots[id]
			if r == nil {
				r = &rootInfo{needsProd: map[string]map[string]bool{}, needsUnexp: map[string]bool{}, needsTest: map[string]bool{}, usedBy: map[string]int{}, usedFiles: map[string]bool{}}
				roots[id] = r
			}
			return r
		}
		for _, d := range decls {
			if d.test {
				get(copyRoot(d)).loc += d.loc
			}
		}
		for k, v := range refs {
			fd, td := decls[k.fromDecl], decls[v.to]
			if !fd.test {
				continue
			}
			fr := copyRoot(fd)
			r := get(fr)
			if !td.test {
				if r.needsProd[td.pkg] == nil {
					r.needsProd[td.pkg] = map[string]bool{}
				}
				r.needsProd[td.pkg][k.sym] = true
				if !exportedEnd(k.sym) {
					r.needsUnexp[td.pkg+":"+k.sym] = true
				}
				continue
			}
			tr := copyRoot(td)
			if tr == fr {
				continue
			}
			r.needsTest[tr] = true
			u := get(tr)
			u.usedBy[fd.pkg] += v.sites
			u.usedFiles[fd.file] = true
		}
		td2, _ := os.Create(filepath.Join(*outFlag, "testdecls.tsv"))
		dw := bufio.NewWriter(td2)
		fmt.Fprintf(dw, "decl\tkind\tfile\tloc\tpkg\tneeds_prod\tneeds_unexported\tneeds_test\tused_by\tused_files\n")
		var rnames []string
		for id := range roots {
			rnames = append(rnames, id)
		}
		sort.Strings(rnames)
		for _, id := range rnames {
			r := roots[id]
			d := decls[id]
			if d == nil || !d.test {
				continue
			}
			var np, ub []string
			for p, m := range r.needsProd {
				np = append(np, fmt.Sprintf("%s=%d", p, len(m)))
			}
			sort.Strings(np)
			for _, e := range sorted(r.usedBy) {
				ub = append(ub, fmt.Sprintf("%s=%d", e.k, e.v))
			}
			fmt.Fprintf(dw, "%s\t%s\t%s\t%d\t%s\t%s\t%s\t%s\t%s\t%d\n", id, d.kind, d.file, r.loc, d.pkg,
				strings.Join(np, " "), strings.Join(keys(boolInt(r.needsUnexp)), " "), strings.Join(keys(boolInt(r.needsTest)), " "),
				strings.Join(ub, " "), len(r.usedFiles))
		}
		dw.Flush()
		td2.Close()
	}

	// ---- external users ----
	pf("\n## 7. external users of %s, by end-state package (distinct identifiers / sites)\n", syncRel)
	syncImp, err := imp.Import(syncPath)
	if err != nil {
		pf("cannot import sync export data: %v\n", err)
		return
	}
	rx := newResolver(syncImp)
	xt, _ := os.Create(filepath.Join(*outFlag, "external.tsv"))
	xtw := bufio.NewWriter(xt)
	fmt.Fprintf(xtw, "user_pkg\ttest\tend_pkg\tsymbol\tsites\n")
	for _, p := range all {
		if p.ImportPath == syncPath {
			continue
		}
		imports := false
		for _, lst := range [][]string{p.Imports, p.TestImports, p.XTestImports} {
			for _, i := range lst {
				if i == syncPath {
					imports = true
				}
			}
		}
		if !imports {
			continue
		}
		short := strings.TrimPrefix(p.ImportPath, modPrefix)
		byPkg := map[bool]map[string]map[string]int{false: {}, true: {}}
		record := func(files []*ast.File, path string) {
			if len(files) == 0 {
				return
			}
			xinfo := &types.Info{Uses: map[*ast.Ident]types.Object{}}
			xconf := types.Config{Importer: imp, Error: func(error) {}}
			xconf.Check(path, fset, files, xinfo)
			for id, obj := range xinfo.Uses {
				if obj.Pkg() != syncImp {
					continue
				}
				ref, ok := rx.resolve(obj)
				if !ok {
					continue
				}
				d := symDecl(decls, ref.name)
				ep := "?"
				if d != nil {
					ep = d.pkg
				}
				isTest := strings.HasSuffix(baseOf(id.Pos()), "_test.go")
				if byPkg[isTest][ep] == nil {
					byPkg[isTest][ep] = map[string]int{}
				}
				byPkg[isTest][ep][ref.name]++
			}
		}
		record(parseFiles(p.Dir, append(append([]string{}, p.GoFiles...), p.TestGoFiles...)), p.ImportPath)
		record(parseFiles(p.Dir, p.XTestGoFiles), p.ImportPath+"_test")
		for _, isTest := range []bool{false, true} {
			if len(byPkg[isTest]) == 0 {
				continue
			}
			label := "production"
			if isTest {
				label = "tests"
			}
			pf("  %-28s %-10s", short, label)
			for _, ep := range append(append([]string{}, cfg.order...), "?") {
				if m := byPkg[isTest][ep]; len(m) > 0 {
					pf(" %s=%d/%d", ep, len(m), sum(m))
					for _, e := range sorted(m) {
						fmt.Fprintf(xtw, "%s\t%v\t%s\t%s\t%d\n", short, isTest, ep, e.k, e.v)
					}
				}
			}
			pf("\n")
		}
	}
	xtw.Flush()
	xt.Close()
	w.Flush()
	fmt.Printf("wrote %s (typecheck errors %d, violations %d, cycles %d, interface problems %d, split const blocks %d, name collisions %d, illegal test references %d)\n",
		repF.Name(), nerr, len(viols), len(cyc), nIfaceBad, nSplitConst, nColl, len(tviols))
}

// walkNamed calls visit for every named type of pkg that appears in t,
// without descending into the definition of those named types.
func walkNamed(t types.Type, pkg *types.Package, seen map[types.Type]bool, visit func(*types.TypeName)) {
	if t == nil || seen[t] {
		return
	}
	seen[t] = true
	switch u := t.(type) {
	case *types.Alias:
		if u.Obj().Pkg() == pkg {
			visit(u.Obj())
		}
		walkNamed(types.Unalias(u), pkg, seen, visit)
	case *types.Named:
		if u.Obj().Pkg() == pkg {
			visit(u.Origin().Obj())
		}
		if ta := u.TypeArgs(); ta != nil {
			for i := 0; i < ta.Len(); i++ {
				walkNamed(ta.At(i), pkg, seen, visit)
			}
		}
	case *types.Pointer:
		walkNamed(u.Elem(), pkg, seen, visit)
	case *types.Slice:
		walkNamed(u.Elem(), pkg, seen, visit)
	case *types.Array:
		walkNamed(u.Elem(), pkg, seen, visit)
	case *types.Chan:
		walkNamed(u.Elem(), pkg, seen, visit)
	case *types.Map:
		walkNamed(u.Key(), pkg, seen, visit)
		walkNamed(u.Elem(), pkg, seen, visit)
	case *types.Signature:
		walkNamed(u.Params(), pkg, seen, visit)
		walkNamed(u.Results(), pkg, seen, visit)
	case *types.Tuple:
		for i := 0; i < u.Len(); i++ {
			walkNamed(u.At(i).Type(), pkg, seen, visit)
		}
	case *types.Struct:
		for i := 0; i < u.NumFields(); i++ {
			walkNamed(u.Field(i).Type(), pkg, seen, visit)
		}
	case *types.Interface:
		for i := 0; i < u.NumMethods(); i++ {
			walkNamed(u.Method(i).Type(), pkg, seen, visit)
		}
	}
}

// exportedName is the identifier a symbol gets when it is exported.
func exportedName(sym string) string {
	i := strings.LastIndex(sym, ".") + 1
	if i >= len(sym) {
		return sym
	}
	return sym[:i] + strings.ToUpper(sym[i:i+1]) + sym[i+1:]
}

func boolInt(m map[string]bool) map[string]int {
	out := map[string]int{}
	for k := range m {
		out[k] = 1
	}
	return out
}

// symDecl returns the declaration a symbol belongs to: the declaration
// itself for package-level names and concrete methods, the owning type for
// fields and interface methods.
func symDecl(decls map[string]*decl, sym string) *decl {
	if d := decls[sym]; d != nil {
		return d
	}
	if i := strings.Index(sym, "."); i >= 0 {
		return decls[sym[:i]]
	}
	return nil
}
