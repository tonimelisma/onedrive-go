// lintmodel estimates two lint consequences of a package split of
// internal/sync, using the declaration-to-package assignment endstate.go wrote
// (rawdecls.tsv):
//
//  1. wrapcheck: return statements that hand back an error produced by a call
//     which is in the same package today and in another package afterwards.
//     wrapcheck does not report same-package calls, so each of these is a new
//     finding. The model follows wrapcheck's own rules: a returned call, or a
//     returned error identifier traced to its most recent assignment.
//  2. forbidigo/lll/errcheck/mnd and the devtool output-boundary, mutex-doc and
//     test-convention checks: test helpers that move into ordinary
//     (non-_test.go) helper packages lose every _test.go exemption; count the
//     raw os.* calls, time.Sleep calls, lines over 140 columns, defer
//     statements, number literals, output-boundary selectors, undocumented
//     mutexes and package-level vars they contain.
//
//	go -C <repo> run /path/lintmodel.go -repo <repo> -decls out/rawdecls.tsv -out out/wrapcheck.tsv
//
// Stdlib only.
package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"flag"
	"fmt"
	"go/ast"
	"go/importer"
	"go/parser"
	"go/token"
	"go/types"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"regexp"
	"sort"
	"strings"
)

var (
	repoFlag  = flag.String("repo", ".", "repo root")
	declsFlag = flag.String("decls", "rawdecls.tsv", "declaration ledger written by endstate.go")
	outFlag   = flag.String("out", "wrapcheck.tsv", "per-site output")
)

const (
	syncRel = "internal/sync"
	tags    = "e2e e2e_full integration"
)

type goPkg struct {
	Dir, ImportPath      string
	GoFiles, TestGoFiles []string
}

func goCmd(args ...string) []byte {
	cmd := exec.Command("nice", append([]string{"go"}, args...)...)
	cmd.Dir = *repoFlag
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	out, err := cmd.Output()
	if err != nil {
		fmt.Fprintf(os.Stderr, "go %s: %v\n%s\n", strings.Join(args, " "), err, stderr.String())
		os.Exit(1)
	}
	return out
}

func recvBaseName(e ast.Expr) string {
	for {
		switch x := e.(type) {
		case *ast.StarExpr:
			e = x.X
		case *ast.ParenExpr:
			e = x.X
		case *ast.IndexExpr:
			e = x.X
		case *ast.IndexListExpr:
			e = x.X
		case *ast.Ident:
			return x.Name
		default:
			return ""
		}
	}
}

func recvNamed(t types.Type) *types.TypeName {
	for {
		switch u := t.(type) {
		case *types.Pointer:
			t = u.Elem()
			continue
		case *types.Named:
			return u.Origin().Obj()
		case *types.Alias:
			t = types.Unalias(u)
			continue
		}
		return nil
	}
}

func isError(t types.Type) bool { return t != nil && t.String() == "error" }

var ignoredNames = map[string]bool{"Errorf": true, "Wrap": true, "Wrapf": true, "WithMessage": true, "WithMessagef": true, "WithStack": true}

var forbiddenOS = regexp.MustCompile(`^(Open|OpenFile|ReadFile|WriteFile|Create|CreateTemp|Mkdir|MkdirAll|ReadDir|Stat|Lstat|Remove|RemoveAll|Rename|Chmod|Chtimes|Symlink|Readlink|Truncate)$`)

func main() {
	flag.Parse()
	abs, _ := filepath.Abs(*repoFlag)
	*repoFlag = abs

	// ---- end-state package of every declaration ----
	declPkg := map[string]string{}
	declTest := map[string]bool{}
	df, err := os.Open(*declsFlag)
	if err != nil {
		panic(err)
	}
	sc := bufio.NewScanner(df)
	sc.Buffer(make([]byte, 1<<20), 1<<24)
	header := map[string]int{}
	for sc.Scan() {
		cols := strings.Split(sc.Text(), "\t")
		if len(header) == 0 {
			for i, c := range cols {
				header[c] = i
			}
			continue
		}
		declPkg[cols[header["id"]]] = cols[header["pkg"]]
		declTest[cols[header["id"]]] = cols[header["test"]] == "true"
	}
	df.Close()

	// ---- load and type-check internal/sync the way endstate.go does ----
	var syncPkg *goPkg
	dec := json.NewDecoder(bytes.NewReader(goCmd("list", "-json", "-tags", tags, "./...")))
	for {
		var p goPkg
		if err := dec.Decode(&p); err == io.EOF {
			break
		} else if err != nil {
			panic(err)
		}
		if strings.HasSuffix(p.ImportPath, "/"+syncRel) {
			q := p
			syncPkg = &q
		}
	}
	depSet := map[string]bool{}
	for _, l := range strings.Split(string(goCmd("list", "-deps", "-test", "-tags", tags, "-f", "{{.ImportPath}}", "./...")), "\n") {
		l = strings.TrimSpace(l)
		if l == "" || strings.Contains(l, " ") || strings.HasSuffix(l, ".test") || l == "unsafe" || l == "C" {
			continue
		}
		depSet[l] = true
	}
	var depList []string
	for d := range depSet {
		depList = append(depList, d)
	}
	sort.Strings(depList)
	exports := map[string]string{}
	for _, l := range strings.Split(string(goCmd(append([]string{"list", "-export", "-tags", tags, "-f", "{{.ImportPath}}\t{{.Export}}"}, depList...)...)), "\n") {
		parts := strings.Split(l, "\t")
		if len(parts) == 2 && parts[1] != "" {
			exports[parts[0]] = parts[1]
		}
	}
	fset := token.NewFileSet()
	imp := importer.ForCompiler(fset, "gc", func(path string) (io.ReadCloser, error) {
		f, ok := exports[path]
		if !ok {
			return nil, fmt.Errorf("no export data for %s", path)
		}
		return os.Open(f)
	})
	var files []*ast.File
	for _, n := range append(append([]string{}, syncPkg.GoFiles...), syncPkg.TestGoFiles...) {
		f, err := parser.ParseFile(fset, filepath.Join(syncPkg.Dir, n), nil, parser.ParseComments|parser.SkipObjectResolution)
		if err != nil {
			panic(err)
		}
		files = append(files, f)
	}
	info := &types.Info{Uses: map[*ast.Ident]types.Object{}, Defs: map[*ast.Ident]types.Object{}, Types: map[ast.Expr]types.TypeAndValue{}, Selections: map[*ast.SelectorExpr]*types.Selection{}}
	nerr := 0
	conf := types.Config{Importer: imp, Error: func(err error) { nerr++ }}
	pkg, _ := conf.Check(syncPkg.ImportPath, fset, files, info)
	scope := pkg.Scope()

	// owner type of every struct field declared in the package
	fieldOwner := map[*types.Var]*types.TypeName{}
	var walk func(t types.Type, owner *types.TypeName, depth int)
	walk = func(t types.Type, owner *types.TypeName, depth int) {
		if depth > 8 {
			return
		}
		switch u := t.(type) {
		case *types.Struct:
			for i := 0; i < u.NumFields(); i++ {
				fieldOwner[u.Field(i)] = owner
				walk(u.Field(i).Type(), owner, depth+1)
			}
		case *types.Pointer:
			walk(u.Elem(), owner, depth+1)
		case *types.Slice:
			walk(u.Elem(), owner, depth+1)
		case *types.Array:
			walk(u.Elem(), owner, depth+1)
		case *types.Map:
			walk(u.Key(), owner, depth+1)
			walk(u.Elem(), owner, depth+1)
		}
	}
	for _, n := range scope.Names() {
		if tn, ok := scope.Lookup(n).(*types.TypeName); ok && !tn.IsAlias() {
			if named, ok := tn.Type().(*types.Named); ok {
				walk(named.Underlying(), tn, 0)
			}
		}
	}

	// declaration id of the object a call resolves to, "" when it is not a
	// declaration of this package (or is something wrapcheck never reports)
	targetDecl := func(obj types.Object) (id, name string) {
		if obj == nil || obj.Pkg() != pkg {
			return "", ""
		}
		switch o := obj.(type) {
		case *types.Func:
			o = o.Origin()
			sig, _ := o.Type().(*types.Signature)
			if sig != nil && sig.Recv() != nil {
				owner := recvNamed(sig.Recv().Type())
				if owner == nil || owner.Parent() != scope {
					return "", ""
				}
				return owner.Name() + "." + o.Name(), o.Name()
			}
			if o.Parent() == scope {
				return o.Name(), o.Name()
			}
		case *types.Var:
			o = o.Origin()
			if o.IsField() {
				if owner := fieldOwner[o]; owner != nil {
					return owner.Name(), o.Name()
				}
			}
		}
		return "", ""
	}

	type site struct {
		pos              token.Position
		from, to, target string
		test, iface      bool
		how              string
	}
	var sites []site
	base := func(p token.Pos) string { return filepath.Base(fset.Position(p).Filename) }

	// examine one call that produced a returned error
	check := func(fromDecl string, call *ast.CallExpr, at token.Pos, how string) {
		from := declPkg[fromDecl]
		if from == "" {
			return
		}
		var obj types.Object
		viaIface := false
		switch fun := call.Fun.(type) {
		case *ast.Ident:
			// a plain call of a package-level function: becomes pkg.Func afterwards
			obj = info.Uses[fun]
			if f, ok := obj.(*types.Func); !ok || f.Parent() != scope {
				return
			}
		case *ast.SelectorExpr:
			if tv, ok := info.Types[fun.X]; ok && tv.Type != nil {
				if _, isIface := tv.Type.Underlying().(*types.Interface); isIface {
					return // wrapcheck already reports every interface method today
				}
				// the store seen through a consumer-side interface in syncexec
				if tn := recvNamed(tv.Type); tn != nil && tn.Name() == "SyncStore" && from == "syncexec" {
					viaIface = true
				}
			}
			obj = info.Uses[fun.Sel]
		default:
			return
		}
		id, name := targetDecl(obj)
		if id == "" {
			return
		}
		exported := strings.ToUpper(name[:1]) + name[1:]
		if ignoredNames[exported] {
			return
		}
		to := declPkg[id]
		if to == "" || to == from {
			return
		}
		sites = append(sites, site{pos: fset.Position(at), from: from, to: to, target: id, test: strings.HasSuffix(base(at), "_test.go"), iface: viaIface, how: how})
	}

	// helper-package lint exposure
	helperPkgs := map[string]bool{"syncfixture": true, "syncstoretest": true}
	osCalls := map[string]int{}
	sleepCalls := map[string]int{}
	longLines := map[string]int{}
	deferCalls := map[string]int{}
	srcLines := map[string][]string{}
	lineOf := func(p token.Pos) (string, int) {
		ps := fset.Position(p)
		if _, ok := srcLines[ps.Filename]; !ok {
			b, err := os.ReadFile(ps.Filename)
			if err != nil {
				panic(err)
			}
			srcLines[ps.Filename] = strings.Split(string(b), "\n")
		}
		return ps.Filename, ps.Line
	}
	// mnd skips _test.go by default; the devtool output-boundary and mutex-doc
	// checks skip _test.go too. Helper packages are ordinary files.
	magicNumbers := map[string]int{}
	outputBoundary := map[string]int{}
	undocMutex := map[string]int{}
	directT := map[string]int{}
	pkgLevelVars := map[string][]string{}
	mndIgnored := map[string]bool{"0": true, "1": true, "2": true, "3": true, "200": true, "201": true,
		"400": true, "401": true, "403": true, "404": true, "429": true, "500": true}
	mutexDecl := regexp.MustCompile(`\b(?:[A-Za-z_][A-Za-z0-9_]*\.)?(?:RW)?Mutex\b`)
	isMutexDeclaration := func(trimmed string) bool {
		if trimmed == "" || strings.HasPrefix(trimmed, "//") || !mutexDecl.MatchString(trimmed) {
			return false
		}
		if strings.Contains(trimmed, "func ") || strings.Contains(trimmed, "type ") ||
			strings.Contains(trimmed, "(") || strings.Contains(trimmed, "{") {
			return false
		}
		return len(strings.Fields(strings.TrimPrefix(trimmed, "var "))) >= 2
	}
	importedPath := func(x ast.Expr) string {
		id, ok := x.(*ast.Ident)
		if !ok {
			return ""
		}
		if pn, ok := info.Uses[id].(*types.PkgName); ok {
			return pn.Imported().Path()
		}
		return ""
	}
	helperScan := func(id string, n ast.Node, isConst, isPkgVar bool) {
		p := declPkg[id]
		if !helperPkgs[p] || !declTest[id] {
			return
		}
		if isPkgVar {
			pkgLevelVars[p] = append(pkgLevelVars[p], id)
		}
		fname, first := lineOf(n.Pos())
		_, last := lineOf(n.End())
		for l := first; l <= last && l-1 < len(srcLines[fname]); l++ {
			line := srcLines[fname][l-1]
			if len([]rune(line)) > 140 {
				longLines[p]++
			}
			trimmed := strings.TrimSpace(line)
			if isMutexDeclaration(trimmed) {
				documented := false
				if idx := strings.Index(line, "//"); idx >= 0 {
					documented = strings.TrimSpace(line[idx+2:]) != ""
				} else if l >= 2 {
					documented = strings.HasPrefix(strings.TrimSpace(srcLines[fname][l-2]), "//")
				}
				if !documented {
					undocMutex[p]++
				}
			}
		}
		ast.Inspect(n, func(m ast.Node) bool {
			switch x := m.(type) {
			case *ast.BasicLit:
				if !isConst && (x.Kind == token.INT || x.Kind == token.FLOAT) && !mndIgnored[x.Value] {
					magicNumbers[p]++
				}
			case *ast.SelectorExpr:
				switch importedPath(x.X) {
				case "log/slog":
					if x.Sel.Name == "Debug" || x.Sel.Name == "Info" || x.Sel.Name == "Warn" || x.Sel.Name == "Error" {
						outputBoundary[p]++
					}
				case "os":
					if x.Sel.Name == "Stdout" || x.Sel.Name == "Stderr" {
						outputBoundary[p]++
					}
				case "log":
					if strings.HasPrefix(x.Sel.Name, "Print") {
						outputBoundary[p]++
					}
				}
				if rid, ok := x.X.(*ast.Ident); ok && rid.Name == "t" {
					switch x.Sel.Name {
					case "Fatal", "Fatalf", "Error", "Errorf":
						directT[p]++
					}
				}
			case *ast.DeferStmt:
				deferCalls[p]++
			case *ast.CallExpr:
				if sel, ok := x.Fun.(*ast.SelectorExpr); ok {
					if f, ok := info.Uses[sel.Sel].(*types.Func); ok && f.Pkg() != nil {
						if f.Pkg().Path() == "os" && forbiddenOS.MatchString(f.Name()) {
							osCalls[p]++
						}
						if f.Pkg().Path() == "time" && f.Name() == "Sleep" {
							sleepCalls[p]++
						}
					}
				}
			}
			return true
		})
	}

	for _, f := range files {
		// every assignment in the file whose left side holds an error identifier
		type assign struct {
			stmt *ast.AssignStmt
			objs []types.Object
		}
		var assigns []assign
		ast.Inspect(f, func(n ast.Node) bool {
			if a, ok := n.(*ast.AssignStmt); ok {
				var objs []types.Object
				for _, lhs := range a.Lhs {
					id, ok := lhs.(*ast.Ident)
					if !ok {
						continue
					}
					if tv, ok := info.Types[lhs]; ok && !isError(tv.Type) {
						continue
					}
					obj := info.Defs[id]
					if obj == nil {
						obj = info.Uses[id]
					}
					if obj != nil && isError(obj.Type()) {
						objs = append(objs, obj)
					}
				}
				if len(objs) > 0 {
					assigns = append(assigns, assign{a, objs})
				}
			}
			return true
		})
		sort.Slice(assigns, func(i, j int) bool { return assigns[i].stmt.Pos() < assigns[j].stmt.Pos() })

		scanBody := func(id string, body ast.Node) {
			ast.Inspect(body, func(n ast.Node) bool {
				ret, ok := n.(*ast.ReturnStmt)
				if !ok {
					return true
				}
				for _, expr := range ret.Results {
					if call, ok := expr.(*ast.CallExpr); ok {
						t := info.Types[expr].Type
						if isError(t) {
							check(id, call, call.Pos(), "returned call")
							break
						}
						if tup, ok := t.(*types.Tuple); ok {
							for i := 0; i < tup.Len(); i++ {
								if isError(tup.At(i).Type()) {
									check(id, call, call.Pos(), "returned call")
									break
								}
							}
						}
						break
					}
					tv, ok := info.Types[expr]
					if !ok || !isError(tv.Type) {
						continue
					}
					ident, ok := expr.(*ast.Ident)
					if !ok {
						break
					}
					obj := info.Uses[ident]
					if obj == nil {
						break
					}
					var recent *ast.AssignStmt
					for _, a := range assigns {
						if a.stmt.Pos() > ident.Pos() {
							break
						}
						for _, o := range a.objs {
							if o == obj {
								recent = a.stmt
							}
						}
					}
					if recent == nil || len(recent.Rhs) == 0 {
						break
					}
					call, ok := recent.Rhs[0].(*ast.CallExpr)
					if !ok {
						break
					}
					check(id, call, ident.Pos(), "returned variable")
				}
				return true
			})
		}

		for _, d := range f.Decls {
			switch x := d.(type) {
			case *ast.FuncDecl:
				id := x.Name.Name
				if x.Recv != nil && len(x.Recv.List) > 0 {
					id = recvBaseName(x.Recv.List[0].Type) + "." + x.Name.Name
				}
				helperScan(id, x, false, false)
				if x.Body != nil {
					scanBody(id, x.Body)
				}
			case *ast.GenDecl:
				for _, sp := range x.Specs {
					switch s := sp.(type) {
					case *ast.ValueSpec:
						if len(s.Names) > 0 {
							helperScan(s.Names[0].Name, sp, x.Tok == token.CONST, x.Tok == token.VAR)
							scanBody(s.Names[0].Name, sp)
						}
					case *ast.TypeSpec:
						helperScan(s.Name.Name, sp, false, false)
					}
				}
			}
		}
	}

	// ---- output ----
	of, err := os.Create(*outFlag)
	if err != nil {
		panic(err)
	}
	w := bufio.NewWriter(of)
	fmt.Fprintln(w, "file\tline\tfrom_pkg\tto_pkg\ttarget\ttest\thow\tvia_interface")
	sort.Slice(sites, func(i, j int) bool {
		if sites[i].pos.Filename != sites[j].pos.Filename {
			return sites[i].pos.Filename < sites[j].pos.Filename
		}
		return sites[i].pos.Line < sites[j].pos.Line
	})
	type edge struct {
		from, to string
		test     bool
	}
	count := map[edge]int{}
	total := map[bool]int{}
	iface := 0
	for _, s := range sites {
		fmt.Fprintf(w, "%s\t%d\t%s\t%s\t%s\t%v\t%s\t%v\n", filepath.Base(s.pos.Filename), s.pos.Line, s.from, s.to, s.target, s.test, s.how, s.iface)
		count[edge{s.from, s.to, s.test}]++
		total[s.test]++
		if s.iface {
			iface++
		}
	}
	w.Flush()
	of.Close()

	fmt.Printf("typecheck errors %d\n", nerr)
	fmt.Printf("wrapcheck: new findings in production code %d, in test code %d (of which through the syncexec store interface %d)\n", total[false], total[true], iface)
	var edges []edge
	for e := range count {
		edges = append(edges, e)
	}
	sort.Slice(edges, func(i, j int) bool {
		if edges[i].test != edges[j].test {
			return !edges[i].test
		}
		if edges[i].from != edges[j].from {
			return edges[i].from < edges[j].from
		}
		return edges[i].to < edges[j].to
	})
	for _, e := range edges {
		kind := "production"
		if e.test {
			kind = "test"
		}
		fmt.Printf("  %-10s %-13s -> %-13s %d\n", kind, e.from, e.to, count[e])
	}
	for p := range helperPkgs {
		fmt.Printf("helper package %s: raw os.* calls %d, time.Sleep calls %d, lines over 140 columns %d, defer statements %d\n", p, osCalls[p], sleepCalls[p], longLines[p], deferCalls[p])
		sort.Strings(pkgLevelVars[p])
		fmt.Printf("helper package %s: number literals outside const declarations and mnd's ignore list %d (upper bound for mnd), output-boundary selectors %d, undocumented mutexes %d, direct t.Fatal/Error calls %d, package-level vars %d %v\n",
			p, magicNumbers[p], outputBoundary[p], undocMutex[p], directT[p], len(pkgLevelVars[p]), pkgLevelVars[p])
	}
}
